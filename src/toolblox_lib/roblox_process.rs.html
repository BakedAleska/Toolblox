1//! Running Roblox clients: finding them, setting their priority and CPU cores, and tiling their

4//! Uses operating-system process and window APIs only. Nothing is read from or written to a

13/// Scheduling priority for a Roblox client.
14#[derive(Debug, Clone, Copy, Default, Serialize, Deserialize, PartialEq, Eq)]
15#[serde(rename_all = "snake_case")]
16pub enum ClientPriority {
17    #[default]
18    Normal,
19    BelowNormal,
20    Low,
21}

23/// Window size used when tiling Roblox clients.
24#[derive(Debug, Clone, Copy, Default, Serialize, Deserialize, PartialEq, Eq)]
25#[serde(rename_all = "snake_case")]
26pub enum TileSize {
27    /// Divide the screen evenly between the windows.
28    #[default]
29    Fit,
30    /// 960 × 540 windows.
31    Medium,
32    /// 640 × 360 windows.
33    Small,
34}

36/// Largest CPU core limit accepted per client.
37pub const MAX_CPU_CORES: u32 = 64;

39/// Seconds to watch for a launched client's process.
40const WATCH_SECONDS: u32 = 30;

42/// How a launched client is tuned. The default leaves the process unchanged.
43#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
44pub struct Tuning {
45    pub priority: ClientPriority,
46    /// Logical cores the client may use. `0` allows every core.
47    pub cpu_cores: u32,
48}
49
50impl Tuning {
51    /// Whether the process needs to be changed at all.
52    pub fn changes_process(&self) -> bool {
53        *self != Self::default()
54    }
55}

57/// Tracks which Roblox processes Toolblox already tuned, so batch launches tune each new client
58/// once.
59#[derive(Debug, Default)]
60pub struct ClientTuner {
61    /// Process IDs claimed by a launch, with the account they belong to.
62    claimed: Mutex<HashMap<u32, u64>>,
63    /// Rotates which cores each limited client gets.
64    next_slot: AtomicUsize,
65}
66
67impl ClientTuner {
68    /// Claims a Roblox process that started after `before` and isn't claimed yet.
69    fn claim(&self, before: &HashSet<u32>, running: &HashSet<u32>, account_id: u64) -> Option<u32> {
70        let mut claimed = self.claimed.lock().ok()?;
71        claimed.retain(|pid, _| running.contains(pid));
72        let pid = running
73            .iter()
74            .copied()
75            .filter(|pid| !before.contains(pid) && !claimed.contains_key(pid))
76            .min()?;
77        claimed.insert(pid, account_id);
78        Some(pid)
79    }
80
81    /// Watches for the client a launch starts and tunes it. If Roblox replaces its first process,
82    /// as it does while updating, the replacement is tuned too.
83    ///
84    /// ponytail: launches in a batch are matched to new processes in the order they appear. Two
85    /// clients that start out of order swap tuning. Matching by command line needs process APIs
86    /// Toolblox doesn't use yet.
87    pub async fn watch(&self, before: HashSet<u32>, tuning: Tuning, account_id: u64) {
88        let slot = self.next_slot.fetch_add(1, Ordering::Relaxed);
89        let mask = affinity_mask(slot, tuning.cpu_cores, available_cores());
90        let mut current = None;
91        for _ in 0..WATCH_SECONDS {
92            tokio::time::sleep(std::time::Duration::from_secs(1)).await;
93            let Ok(Ok(running)) = tauri::async_runtime::spawn_blocking(roblox_pids).await else {
94                continue;
95            };
96            if !current.is_some_and(|pid| running.contains(&pid)) {
97                current = self.claim(&before, &running, account_id);
98                if let Some(pid) = current
99                    && (tuning.priority != ClientPriority::Normal || mask.is_some())
100                {
101                    match apply(pid, tuning.priority, mask) {
102                        Ok(()) => log::info!("Tuned Roblox client {pid} for account {account_id}"),
103                        Err(error) => log::warn!("Roblox client {pid} couldn't be tuned: {error}"),
104                    }
105                }
106            }
107        }
108        if current.is_none() {
109            log::info!("No new Roblox client appeared for account {account_id}");
110        }
111    }
112}

114/// Number of logical cores, capped at the 64 one affinity mask can hold.
115fn available_cores() -> u32 {
116    std::thread::available_parallelism()
117        .map_or(1, |count| count.get() as u32)
118        .min(usize::BITS)
119}
120
121/// The cores client number `slot` may use: `cores` consecutive cores, starting where the previous
122/// client's left off. Returns `None` when every core is allowed.
123pub(crate) fn affinity_mask(slot: usize, cores: u32, available: u32) -> Option<usize> {
124    if cores == 0 || cores >= available {
125        return None;
126    }
127    let start = (slot * cores as usize) % available as usize;
128    Some((0..cores as usize).fold(0, |mask, offset| {
129        mask | 1 << ((start + offset) % available as usize)
130    }))
131}

133/// Process IDs of the running Roblox clients.
134pub fn roblox_pids() -> Result<HashSet<u32>, String> {
135    #[cfg(windows)]
136    {
137        use std::os::windows::process::CommandExt;
138
139        const CREATE_NO_WINDOW: u32 = 0x08000000;
140        let output = std::process::Command::new("tasklist.exe")
141            .args([
142                "/FI",
143                "IMAGENAME eq RobloxPlayerBeta.exe",
144                "/FO",
145                "CSV",
146                "/NH",
147            ])
148            .creation_flags(CREATE_NO_WINDOW)
149            .output()
150            .map_err(|_| {
151                "Toolblox couldn't inspect running Roblox processes. Can you try again?".to_string()
152            })?;
153        Ok(parse_tasklist(&String::from_utf8_lossy(&output.stdout)))
154    }
155    #[cfg(target_os = "macos")]
156    {
157        let output = std::process::Command::new("/usr/bin/pgrep")
158            .args(["-x", "RobloxPlayer"])
159            .output()
160            .map_err(|_| {
161                "Toolblox couldn't inspect running Roblox processes. Can you try again?".to_string()
162            })?;
163        Ok(String::from_utf8_lossy(&output.stdout)
164            .lines()
165            .filter_map(|line| line.trim().parse().ok())
166            .collect())
167    }
168    #[cfg(not(any(windows, target_os = "macos")))]
169    {
170        Ok(HashSet::new())
171    }
172}

174/// Reads process IDs from `tasklist /FO CSV /NH` output.
175#[cfg_attr(not(windows), allow(dead_code))]
176fn parse_tasklist(text: &str) -> HashSet<u32> {
177    text.lines()
178        .filter_map(|line| line.split("\",\"").nth(1))
179        .filter_map(|value| value.trim_matches('"').parse().ok())
180        .collect()
181}

183/// Sets a client's priority and, on Windows, the cores it may use.
184fn apply(pid: u32, priority: ClientPriority, mask: Option<usize>) -> Result<(), String> {
185    #[cfg(windows)]
186    {
187        use windows_sys::Win32::Foundation::CloseHandle;
188        use windows_sys::Win32::System::Threading::{
189            BELOW_NORMAL_PRIORITY_CLASS, IDLE_PRIORITY_CLASS, OpenProcess, PROCESS_SET_INFORMATION,
190            SetPriorityClass, SetProcessAffinityMask,
191        };
192
193        // SAFETY: `OpenProcess` takes no pointers; a null result is handled.
194        let process = unsafe { OpenProcess(PROCESS_SET_INFORMATION, 0, pid) };
195        if process.is_null() {
196            return Err("The Roblox client couldn't be opened. Has it closed?".into());
197        }
198        let class = match priority {
199            ClientPriority::Normal => None,
200            ClientPriority::BelowNormal => Some(BELOW_NORMAL_PRIORITY_CLASS),
201            ClientPriority::Low => Some(IDLE_PRIORITY_CLASS),
202        };
203        // SAFETY: `process` is an open handle with set-information access, closed once below.
204        let applied = unsafe {
205            class.is_none_or(|class| SetPriorityClass(process, class) != 0)
206                && mask.is_none_or(|mask| SetProcessAffinityMask(process, mask) != 0)
207        };
208        // SAFETY: `process` came from `OpenProcess` and isn't used again.
209        unsafe { CloseHandle(process) };
210        if applied {
211            Ok(())
212        } else {
213            Err("The Roblox client's priority couldn't be changed. Has it closed?".into())
214        }
215    }
216    #[cfg(target_os = "macos")]
217    {
218        unsafe extern "C" {
219            fn setpriority(
220                which: std::ffi::c_int,
221                who: u32,
222                priority: std::ffi::c_int,
223            ) -> std::ffi::c_int;
224        }
225        const PRIO_PROCESS: std::ffi::c_int = 0;
226        let _ = mask;
227        let nice = match priority {
228            ClientPriority::Normal => return Ok(()),
229            ClientPriority::BelowNormal => 5,
230            ClientPriority::Low => 15,
231        };
232        // SAFETY: `setpriority` takes no pointers.
233        if unsafe { setpriority(PRIO_PROCESS, pid, nice) } == 0 {
234            Ok(())
235        } else {
236            Err("The Roblox client's priority couldn't be changed. Has it closed?".into())
237        }
238    }
239    #[cfg(not(any(windows, target_os = "macos")))]
240    {
241        let _ = (pid, priority, mask);
242        Err("Changing client priority isn't supported on this platform.".into())
243    }
244}

246/// A screen rectangle in physical pixels.
247#[cfg(any(windows, test))]
248#[derive(Debug, Clone, Copy, PartialEq, Eq)]
249pub(crate) struct Rect {
250    pub x: i32,
251    pub y: i32,
252    pub width: i32,
253    pub height: i32,
254}
255
256/// Offset between layers when fixed-size windows don't all fit on the screen.
257#[cfg(any(windows, test))]
258const LAYER_OFFSET: i32 = 32;
259
260/// Positions for `count` windows tiled in `area`, left to right and top to bottom.
261#[cfg(any(windows, test))]
262pub(crate) fn tile_rects(count: usize, area: Rect, size: TileSize) -> Vec<Rect> {
263    if count == 0 || area.width <= 0 || area.height <= 0 {
264        return Vec::new();
265    }
266    let (width, height) = match size {
267        TileSize::Fit => {
268            let columns = (count as f64).sqrt().ceil() as usize;
269            let rows = count.div_ceil(columns);
270            (area.width / columns as i32, area.height / rows as i32)
271        }
272        TileSize::Medium => (960.min(area.width), 540.min(area.height)),
273        TileSize::Small => (640.min(area.width), 360.min(area.height)),
274    };
275    let columns = (area.width / width).max(1) as usize;
276    let rows = (area.height / height).max(1) as usize;
277    (0..count)
278        .map(|index| {
279            let layer = (index / (columns * rows)) as i32;
280            let cell = index % (columns * rows);
281            Rect {
282                x: area.x + (cell % columns) as i32 * width + layer * LAYER_OFFSET,
283                y: area.y + (cell / columns) as i32 * height + layer * LAYER_OFFSET,
284                width,
285                height,
286            }
287        })
288        .collect()
289}

291/// The visible, unowned top-level windows of the given processes, sorted by process ID.
292#[cfg(windows)]
293fn client_windows(pids: HashSet<u32>) -> Vec<(u32, windows_sys::Win32::Foundation::HWND)> {
294    use windows_sys::Win32::Foundation::{HWND, LPARAM};
295    use windows_sys::Win32::UI::WindowsAndMessaging::{
296        EnumWindows, GW_OWNER, GetWindow, GetWindowThreadProcessId, IsWindowVisible,
297    };
298
299    struct Search {
300        pids: HashSet<u32>,
301        windows: Vec<(u32, HWND)>,
302    }
303
304    unsafe extern "system" fn visit(window: HWND, search: LPARAM) -> i32 {
305        // SAFETY: `search` is the `Search` passed to `EnumWindows`, alive for the whole call.
306        let search = unsafe { &mut *(search as *mut Search) };
307        let mut pid = 0;
308        // SAFETY: `window` comes from `EnumWindows` and `pid` is a valid out pointer.
309        unsafe {
310            GetWindowThreadProcessId(window, &mut pid);
311            if search.pids.contains(&pid)
312                && IsWindowVisible(window) != 0
313                && GetWindow(window, GW_OWNER).is_null()
314            {
315                search.windows.push((pid, window));
316            }
317        }
318        1
319    }
320
321    let mut search = Search {
322        pids,
323        windows: Vec::new(),
324    };
325    if !search.pids.is_empty() {
326        // SAFETY: `visit` only casts the `LPARAM` back to `search`, which outlives the call.
327        unsafe { EnumWindows(Some(visit), &mut search as *mut Search as LPARAM) };
328    }
329    search.windows.sort_by_key(|(pid, _)| *pid);
330    search.windows
331}

333/// Restores a window from minimized or maximized, then gives it the rectangle.
334#[cfg(windows)]
335fn place_window(window: windows_sys::Win32::Foundation::HWND, rect: Rect) {
336    use windows_sys::Win32::UI::WindowsAndMessaging::{
337        IsIconic, IsZoomed, SW_RESTORE, SWP_NOACTIVATE, SWP_NOZORDER, SetWindowPos, ShowWindow,
338    };
339
340    // SAFETY: `window` is a top-level window found by `client_windows`. A window that closed since
341    // then makes these calls fail harmlessly.
342    unsafe {
343        if IsIconic(window) != 0 || IsZoomed(window) != 0 {
344            ShowWindow(window, SW_RESTORE);
345        }
346        SetWindowPos(
347            window,
348            std::ptr::null_mut(),
349            rect.x,
350            rect.y,
351            rect.width,
352            rect.height,
353            SWP_NOZORDER | SWP_NOACTIVATE,
354        );
355    }
356}

358/// Tiles every visible Roblox client window on the primary screen and returns how many moved.
359pub fn tile_windows(size: TileSize) -> Result<usize, String> {
360    #[cfg(windows)]
361    {
362        use windows_sys::Win32::Foundation::RECT;
363        use windows_sys::Win32::UI::WindowsAndMessaging::{SPI_GETWORKAREA, SystemParametersInfoW};
364
365        let windows = client_windows(roblox_pids()?);
366        if windows.is_empty() {
367            return Ok(0);
368        }
369        let mut work = RECT {
370            left: 0,
371            top: 0,
372            right: 0,
373            bottom: 0,
374        };
375        // SAFETY: `SPI_GETWORKAREA` writes one `RECT` to the pointer.
376        let found = unsafe {
377            SystemParametersInfoW(SPI_GETWORKAREA, 0, (&mut work as *mut RECT).cast(), 0)
378        };
379        if found == 0 {
380            return Err("The screen size couldn't be read. Can you try again?".into());
381        }
382        let area = Rect {
383            x: work.left,
384            y: work.top,
385            width: work.right - work.left,
386            height: work.bottom - work.top,
387        };
388        let rects = tile_rects(windows.len(), area, size);
389        for ((_, window), rect) in windows.iter().zip(&rects) {
390            place_window(*window, *rect);
391        }
392        Ok(rects.len())
393    }
394    #[cfg(not(windows))]
395    {
396        let _ = size;
397        Err("Tiling windows isn't supported on macOS yet.".into())
398    }
