1//! Toolblox native host.
2//!
3//! Toolblox is a Tauri 2 desktop app for managing and launching Roblox accounts. This crate owns
4//! everything that touches secrets, the network, the file system, or other processes. The React
5//! interface receives display data and opaque IDs only.
6//!
7//! # Modules
8//!
9//! - [`app_commands`]: the Tauri commands the interface calls, and the shared [`Runtime`].
10//! - [`startup`] and [`update`]: the fail-closed update gate that runs before the app opens.
11//! - [`storage`] and [`models`]: validated, atomic JSON persistence for settings and accounts.
12//! - [`credentials`]: Roblox sessions in Windows Credential Manager or the macOS Keychain.
13//! - [`login`]: the isolated Roblox sign-in window.
14//! - [`roblox`]: Roblox web APIs for profiles, presence, games, and join tickets.
15//! - [`auto_rejoin`]: presence monitoring for drop notifications and auto-rejoin.
16//! - [`multi_instance`]: running several Roblox clients on Windows and macOS.
17//! - [`roblox_process`]: client priority, CPU cores, and window tiling.
18//! - [`roblox_files`]: the advanced settings that write Roblox's files, and the only code that does.
19//! - [`lifecycle`]: window, tray, and single-instance behavior.
20//! - [`logs`]: the rotating, redacted log file.
21//! - [`widgets`], [`widget_protocol`], [`widget_ipc`], and [`widget_process`]: signed widget
22//!   packages, their assets, permission-checked messages, and sidecar processes.
23//!
24//! # Data
25//!
26//! Data lives in the Toolblox data folder: `%LOCALAPPDATA%\Toolblox` on Windows and
27//! `~/Library/Application Support/Toolblox` on macOS.

58/// Returns the Toolblox data folder.
59///
60/// This is `%LOCALAPPDATA%\Toolblox` on Windows and `~/Library/Application Support/Toolblox` on
61/// macOS. Debug builds use `TOOLBLOX_DATA_DIR` instead when it's set.
62fn toolblox_data_root() -> Result<PathBuf, std::io::Error> {
63    #[cfg(debug_assertions)]
64    if let Some(path) = std::env::var_os("TOOLBLOX_DATA_DIR") {
65        return Ok(PathBuf::from(path));
66    }
67    #[cfg(windows)]
68    let base = std::env::var_os("LOCALAPPDATA")
69        .map(PathBuf::from)
70        .or_else(|| {
71            std::env::var_os("USERPROFILE").map(|home| PathBuf::from(home).join("AppData/Local"))
72        });
73    #[cfg(target_os = "macos")]
74    let base = std::env::var_os("HOME")
75        .map(|home| PathBuf::from(home).join("Library/Application Support"));
76    #[cfg(not(any(windows, target_os = "macos")))]
77    let base = std::env::var_os("XDG_DATA_HOME")
78        .map(PathBuf::from)
79        .or_else(|| std::env::var_os("HOME").map(|home| PathBuf::from(home).join(".local/share")));
80    base.map(|path| path.join("Toolblox")).ok_or_else(|| {
81        std::io::Error::new(
82            std::io::ErrorKind::NotFound,
83            "Toolblox couldn't locate its data folder. Is the user profile available?",
84        )
85    })
86}

88/// Moves data that 0.2.0 beta saved in a `v2` subfolder into the data folder.
89///
90/// Moved entries replace same-named ones already in the data folder. An interrupted move
91/// continues on the next launch, and nothing happens once the subfolder is gone.
92fn move_beta_data(data_root: &std::path::Path) -> std::io::Result<()> {
93    let beta_root = data_root.join("v2");
94    let entries = match std::fs::read_dir(&beta_root) {
95        Ok(entries) => entries,
96        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(()),
97        Err(error) => return Err(error),
98    };
99    for entry in entries {
100        let entry = entry?;
101        let target = data_root.join(entry.file_name());
102        match std::fs::symlink_metadata(&target) {
103            Ok(metadata) if metadata.is_dir() => std::fs::remove_dir_all(&target)?,
104            Ok(_) => std::fs::remove_file(&target)?,
105            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
106            Err(error) => return Err(error),
107        }
108        std::fs::rename(entry.path(), target)?;
109    }
110    std::fs::remove_dir(beta_root)
111}

113/// Name of the marker file that requests safe mode for the next launch.
114const SAFE_MODE_MARKER: &str = "safe-mode-next-launch";
115
116/// Records whether the next launch skips widgets. Failures only lose the one-time request.
117pub fn set_safe_mode_marker(data_root: &std::path::Path, enabled: bool) {
118    let marker = data_root.join(SAFE_MODE_MARKER);
119    let _ = if enabled {
120        std::fs::create_dir_all(data_root).and_then(|_| std::fs::write(marker, b""))
121    } else {
122        std::fs::remove_file(marker)
123    };
124}

126/// Safe mode applies for this launch when requested with `--safe-mode` or by the marker, which is
127/// consumed so the following launch loads widgets again.
128fn take_safe_mode(data_root: &std::path::Path) -> bool {
129    let marker = std::fs::remove_file(data_root.join(SAFE_MODE_MARKER)).is_ok();
130    marker || std::env::args().any(|argument| argument == "--safe-mode")
131}

133/// Starts the processes of widgets marked to start on launch, unless safe mode is active.
134fn start_widgets_on_launch(runtime: &Runtime) {
135    if runtime.safe_mode {
136        return;
137    }
138    let Ok(settings) = runtime.storage.load_settings() else {
139        return;
140    };
141    let Ok(mut processes) = runtime.processes.lock() else {
142        return;
143    };
144    for widget in widgets::discover_installed(&runtime.widgets_root)
145        .into_iter()
146        .filter_map(Result::ok)
147        .filter(|widget| {
148            settings.widgets_start_on_launch.contains(&widget.id)
149                && !settings.disabled_widgets.contains(&widget.id)
150        })
151    {
152        let root = runtime.widgets_root.join(&widget.id);
153        for declaration in &widget.manifest.processes {
154            if let Err(error) = processes.start(&widget.id, &root, declaration, &[]) {
155                log::warn!(
156                    "Widget {} process {} didn't start on launch: {error}",
157                    widget.id,
158                    declaration.id
159                );
160            }
161        }
162    }
163}

165/// Builds and runs the Tauri application.
166#[cfg_attr(mobile, tauri::mobile_entry_point)]
167pub fn run() {
168    let lifecycle_state = Arc::new(LifecycleState::default());
169    let single_instance_state = Arc::clone(&lifecycle_state);
170    tauri::Builder::default()
171        .plugin(tauri_plugin_single_instance::init(move |app, _, _| {
172            lifecycle::activate_or_defer(app, &single_instance_state);
173        }))
174        .plugin(tauri_plugin_opener::init())
175        .plugin(tauri_plugin_notification::init())
176        .plugin(tauri_plugin_updater::Builder::new().build())
177        .register_uri_scheme_protocol("widget", widget_protocol::response)
178        .setup(move |app| {
179            app.handle().plugin(tauri_plugin_autostart::init(
180                tauri_plugin_autostart::MacosLauncher::LaunchAgent,
181                None,
182            ))?;
183
184            let data_root = toolblox_data_root()?;
185            let moved = move_beta_data(&data_root);
186            let widgets_root = data_root.join("widgets");
187            let safe_mode = take_safe_mode(&data_root);
188            let _ = logs::init(&data_root.join("logs"));
189            log::info!(
190                "Toolblox {} ({}) starting on {} {}{}",
191                env!("CARGO_PKG_VERSION"),
192                if cfg!(debug_assertions) {
193                    "debug"
194                } else {
195                    "release"
196                },
197                std::env::consts::OS,
198                std::env::consts::ARCH,
199                if safe_mode { " in safe mode" } else { "" }
200            );
201            if let Err(error) = moved {
202                log::error!("Moving data from the 0.2.0 beta folder failed: {error}");
203            }
204            let storage = Arc::new(storage::Storage::new(data_root.clone()));
205            let credentials = Arc::new(credentials::OsCredentialStore);
206            let client = reqwest::Client::builder()
207                .https_only(true)
208                .redirect(reqwest::redirect::Policy::none())
209                .timeout(std::time::Duration::from_secs(20))
210                .user_agent(concat!("Toolblox/", env!("CARGO_PKG_VERSION")))
211                .build()?;
212            let processes = Arc::new(Mutex::new(widget_process::WidgetProcessManager::default()));
213            let runtime = Runtime {
214                storage,
215                credentials,
216                client,
217                widgets_root,
218                catalogue: Mutex::new(None),
219                catalogue_error: Mutex::new(None),
220                processes: Arc::clone(&processes),
221                widget_sessions: Mutex::new(widget_ipc::WidgetSessions::default()),
222                multi_instance_helper: Some(
223                    app.path()
224                        .resource_dir()?
225                        .join("resources/multi-instance/multi_instance_helper.exe"),
226                ),
227                cleared_roblox_pids: Mutex::new(Default::default()),
228                client_tuner: Default::default(),
229                expired_sessions: Mutex::new(Default::default()),
230                safe_mode,
231            };
232            app.manage(runtime);
233            app.manage(Arc::clone(&lifecycle_state));
234            app.manage(startup::StartupState::new());
235
236            let before_quit = Arc::new(move || {
237                if let Ok(mut processes) = processes.lock() {
238                    processes.stop_all();
239                }
240            });
241            let safe_mode_root = data_root.clone();
242            lifecycle::install_tray(
243                app.handle(),
244                Arc::clone(&lifecycle_state),
245                before_quit,
246                Arc::new(move || set_safe_mode_marker(&safe_mode_root, true)),
247            )?;
248
249            if let Some(window) = app.get_webview_window(lifecycle::MAIN_WINDOW_LABEL) {
250                let close_state = Arc::clone(&lifecycle_state);
251                let close_window = window.clone();
252                window.on_window_event(move |event| {
253                    if let tauri::WindowEvent::CloseRequested { api, .. } = event {
254                        let app = close_window.app_handle();
255                        let startup_ready = matches!(
256                            app.state::<startup::StartupState>().view(),
257                            startup::StartupView::Ready
258                        );
259                        let run_in_background = startup_ready
260                            && app
261                                .state::<Runtime>()
262                                .storage
263                                .load_settings()
264                                .map(|settings| settings.run_in_background)
265                                .unwrap_or(false);
266                        if matches!(
267                            lifecycle::handle_close_request(
268                                &close_window,
269                                api,
270                                &close_state,
271                                run_in_background,
272                            ),
273                            Ok(lifecycle::CloseDecision::Exit)
274                        ) {
275                            app.exit(0);
276                        }
277                    }
278                });
279            }
280
281            if let Some(window) = app.get_webview_window(lifecycle::MAIN_WINDOW_LABEL) {
282                window.show()?;
283                lifecycle::apply_pending_activation(app.handle(), &lifecycle_state)?;
284            }
285            tauri::async_runtime::spawn(startup::run(app.handle().clone()));
286            let services_app = app.handle().clone();
287            tauri::async_runtime::spawn(async move {
288                startup::wait_until_ready(&services_app).await;
289                #[cfg(not(debug_assertions))]
290                {
291                    use tauri_plugin_autostart::ManagerExt;
292
293                    let runtime = services_app.state::<Runtime>();
294                    if let Ok(settings) = runtime.storage.load_settings() {
295                        let autostart = services_app.autolaunch();
296                        if autostart.is_enabled().ok() != Some(settings.open_on_launch) {
297                            let _ = if settings.open_on_launch {
298                                autostart.enable()
299                            } else {
300                                autostart.disable()
301                            };
302                        }
303                    }
304                }
305                let always_on_top = services_app
306                    .state::<Runtime>()
307                    .storage
308                    .load_settings()
309                    .is_ok_and(|settings| settings.always_on_top);
310                if always_on_top
311                    && let Some(window) =
312                        services_app.get_webview_window(lifecycle::MAIN_WINDOW_LABEL)
313                {
314                    let _ = window.set_always_on_top(true);
315                }
316                app_commands::backfill_profiles(&services_app.state::<Runtime>()).await;
317                app_commands::check_sessions(&services_app.state::<Runtime>()).await;
318                start_widgets_on_launch(&services_app.state::<Runtime>());
319                auto_rejoin::run(services_app).await;
320            });
321            Ok(())
322        })
323        .invoke_handler(tauri::generate_handler![
324            app_commands::get_app_state,
325            app_commands::begin_roblox_login,
326            app_commands::update_account_notes,
327            app_commands::set_account_place,
328            app_commands::set_account_priority,
329            app_commands::set_account_launch_settings,
330            app_commands::tile_roblox_windows,
331            app_commands::place_exists,
332            app_commands::forget_recent_place,
333            app_commands::get_widget_store,
334            app_commands::set_widget_store,
335            app_commands::restart_app,
336            app_commands::remove_account,
337            app_commands::join_accounts,
338            app_commands::reorder_accounts,
339            app_commands::save_settings,
340            app_commands::install_widget,
341            app_commands::set_widget_enabled,
342            app_commands::set_widget_start_on_launch,
343            app_commands::update_widget,
344            app_commands::uninstall_widget,
345            app_commands::retry_catalogue,
346            app_commands::open_widget,
347            app_commands::close_widget_session,
348            app_commands::widget_request,
349            app_commands::read_log,
350            app_commands::open_log_folder,
351            app_commands::write_log,
352            app_commands::get_theme,
353            app_commands::save_theme,
354            app_commands::embed_theme_files,
355            app_commands::export_theme,
356            startup::startup_status,
357            startup::retry_startup_update,
358            startup::quit_startup,
359        ])
360        .build(tauri::generate_context!())
361        .expect("Toolblox couldn't start")
362        .run(|app, event| {
363            #[cfg(target_os = "macos")]
364            if matches!(event, tauri::RunEvent::Reopen { .. }) {
365                lifecycle::activate_or_defer(
366                    app,
367                    app.state::<Arc<LifecycleState>>().inner().as_ref(),
368                );
369            }
370            if matches!(event, tauri::RunEvent::Exit) {
371                if let Ok(mut processes) = app.state::<Runtime>().processes.lock() {
372                    processes.stop_all();
373                }
374                log::info!("Toolblox exited");
375                log::logger().flush();
376            }
377        });
378}

384    /// Module widgets share the main webview, so its permissions and network policy are pinned.
385    /// Widening them needs security review: no permission may create or script webviews, read
386    /// cookies, or reach the network outside IPC.
387    #[test]
388    fn webview_permissions_match_the_reviewed_list() {
389        let capability: serde_json::Value =
390            serde_json::from_str(include_str!("../capabilities/default.json")).unwrap();
391        assert_eq!(
392            capability["permissions"],
393            serde_json::json!(["core:default", "opener:default"])
394        );
395        let capabilities = std::fs::read_dir(concat!(env!("CARGO_MANIFEST_DIR"), "/capabilities"))
396            .unwrap()
397            .count();
398        assert_eq!(capabilities, 1);
399
400        let config: serde_json::Value =
401            serde_json::from_str(include_str!("../tauri.conf.json")).unwrap();
402        let security = &config["app"]["security"];
403        assert_eq!(security["capabilities"], serde_json::json!(["default"]));
404        let connect = security["csp"]
405            .as_str()
406            .unwrap()
407            .split(';')
408            .map(str::trim)
409            .find(|directive| directive.starts_with("connect-src"));
410        assert_eq!(connect, Some("connect-src ipc: http://ipc.localhost"));
411    }

413    #[test]
414    fn beta_data_moves_up_and_replaces_older_entries() {
415        let root = tempfile::tempdir().unwrap();
416        let data = root.path();
417        std::fs::create_dir_all(data.join("v2/logs")).unwrap();
418        std::fs::create_dir_all(data.join("logs")).unwrap();
419        std::fs::write(data.join("v2/accounts.json"), "new").unwrap();
420        std::fs::write(data.join("v2/logs/toolblox.log"), "new").unwrap();
421        std::fs::write(data.join("accounts.json"), "old").unwrap();
422        std::fs::write(data.join("logs/old.log"), "old").unwrap();
423        std::fs::write(data.join("notes.txt"), "kept").unwrap();
424
425        move_beta_data(data).unwrap();
426
427        assert!(!data.join("v2").exists());
428        assert_eq!(
429            std::fs::read_to_string(data.join("accounts.json")).unwrap(),
430            "new"
431        );
432        assert!(data.join("logs/toolblox.log").exists());
433        assert!(!data.join("logs/old.log").exists());
434        assert!(data.join("notes.txt").exists());
435        move_beta_data(data).unwrap();
436        assert_eq!(
437            std::fs::read_to_string(data.join("accounts.json")).unwrap(),
438            "new"
439        );
440    }

443    fn safe_mode_marker_applies_to_one_launch() {
444        let root = tempfile::tempdir().unwrap();
