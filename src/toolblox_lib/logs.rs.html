1//! Rotating, redacted log file shared by the host, the interface, and widgets.
2//!
3//! Records go through the [`log`] facade. Toolblox's own records are kept from `Info` up (`Debug`
4//! in debug builds); other crates, such as Tauri and its plugins, from `Warn` up. Every line is
5//! redacted before it's written, so cookies, tickets, authorization headers, and launch URLs never
6//! reach the file, whoever logged them.
7//!
8//! The file is `logs/toolblox.log` in the data folder. It rotates at 1 MB and keeps three older
9//! files, `toolblox.log.1` through `toolblox.log.3`.
10

22/// Name of the current log file.
23const FILE_NAME: &str = "toolblox.log";
24/// Size at which the current file rotates.
25const MAX_FILE_BYTES: u64 = 1_000_000;
26/// Rotated files kept next to the current one.
27const BACKUPS: usize = 3;
28/// Longest message kept per record, in bytes. Longer messages are cut.
29pub const MAX_MESSAGE_BYTES: usize = 8 * 1024;
30/// Most lines [`tail`] returns.
31pub const MAX_TAIL_LINES: usize = 2_000;
32/// Target of records logged from the crate root, shown as `toolblox`.
33const CRATE_ROOT: &str = "toolblox_lib";
34/// Prefix of Toolblox's own module paths, dropped from record targets.
35const CRATE_PREFIX: &str = "toolblox_lib::";
36
37/// Targets for records that don't come from a Toolblox module.
38pub const OWN_TARGETS: &[&str] = &["interface", "panic"];
39/// Prefix of widget record targets, followed by the widget ID.
40pub const WIDGET_TARGET_PREFIX: &str = "widget:";
41
42/// Replacement for redacted values.
43const REDACTED: &str = "[redacted]";

45/// Keys whose values are secret wherever they appear as `key=value`, `key: value`, or
46/// `"key": "value"`. Matched case-insensitively at word boundaries.
47const SECRET_KEYS: &[&str] = &[
48    ".roblosecurity",
49    "rbx-authentication-ticket",
50    "authentication-ticket",
51    "ticket",
52    "x-csrf-token",
53    "authorization",

55    "cookie",
56    "set-cookie",
57    "password",
58    "secret",
59    "token",

61    "refresh_token",
62    "api_key",
63    "apikey",
64];

66/// Keys whose values run to the end of the line or quote, because they can contain spaces or
67/// several `name=value` pairs.
68const WHOLE_VALUE_KEYS: &[&str] = &[
69    "authorization",
70    "proxy-authorization",
71    "cookie",
72    "set-cookie",
73];

75/// Schemes whose URIs carry join tickets. Everything after the scheme is redacted.
76const LAUNCH_SCHEMES: &[&str] = &["roblox-player:", "roblox-studio:"];

80/// Appends records to the current file, rotating it when it grows too large.
81struct FileLogger {
82    directory: PathBuf,
83    file: Mutex<Option<File>>,
84}
85
86impl FileLogger {
87    fn path(&self) -> PathBuf {
88        self.directory.join(FILE_NAME)
89    }
90
91    fn append(&self, line: &str) {
92        let mut file = self.file.lock().unwrap_or_else(|error| error.into_inner());
93        let size = file
94            .as_ref()
95            .and_then(|file| file.metadata().ok())
96            .map_or(0, |metadata| metadata.len());
97        if file.is_none() || size + line.len() as u64 > MAX_FILE_BYTES {
98            *file = None;
99            if size > 0 {
100                rotate(&self.directory);
101            }
102            *file = open(&self.path()).ok();
103        }
104        if let Some(file) = file.as_mut() {
105            let _ = file.write_all(line.as_bytes());
106        }
107    }
108}
109
110impl Log for FileLogger {
111    fn enabled(&self, metadata: &Metadata) -> bool {
112        metadata.level() <= level_for(metadata.target())
113    }
114
115    fn log(&self, record: &Record) {
116        if self.enabled(record.metadata()) {
117            let message = record.args().to_string();
118            self.append(&format_line(
119                SystemTime::now(),
120                record.level(),
121                record.target(),
122                &message,
123            ));
124        }
125    }
126
127    fn flush(&self) {
128        if let Ok(mut file) = self.file.lock()
129            && let Some(file) = file.as_mut()
130        {
131            let _ = file.flush();
132        }
133    }
134}

136/// Highest level kept for a target.
137fn level_for(target: &str) -> LevelFilter {
138    let own = target == CRATE_ROOT
139        || target.starts_with(CRATE_PREFIX)
140        || OWN_TARGETS.contains(&target)
141        || target.starts_with(WIDGET_TARGET_PREFIX);
142    match (own, cfg!(debug_assertions)) {
143        (true, true) => LevelFilter::Debug,
144        (true, false) => LevelFilter::Info,
145        (false, _) => LevelFilter::Warn,
146    }
147}

149/// Opens the log file for appending, creating it if needed.
150fn open(path: &Path) -> io::Result<File> {
151    OpenOptions::new().create(true).append(true).open(path)
152}

154/// Shifts `toolblox.log.N` to `N + 1`, dropping the oldest, then moves the current file to `.1`.
155fn rotate(directory: &Path) {
156    let numbered = |index: usize| directory.join(format!("{FILE_NAME}.{index}"));
157    let _ = fs::remove_file(numbered(BACKUPS));
158    for index in (1..BACKUPS).rev() {
159        let _ = fs::rename(numbered(index), numbered(index + 1));
160    }
161    let _ = fs::rename(directory.join(FILE_NAME), numbered(1));
162}

164/// Starts logging to `directory` and records panics. Later calls do nothing.
165pub fn init(directory: &Path) -> io::Result<()> {
166    fs::create_dir_all(directory)?;
167    let file = open(&directory.join(FILE_NAME))?;
168    let logger = LOGGER.get_or_init(|| FileLogger {
169        directory: directory.to_path_buf(),
170        file: Mutex::new(Some(file)),
171    });
172    if log::set_logger(logger).is_err() {
173        return Ok(());
174    }
175    log::set_max_level(LevelFilter::Debug);
176    let previous = std::panic::take_hook();
177    std::panic::set_hook(Box::new(move |info| {
178        let backtrace = std::backtrace::Backtrace::force_capture();
179        log::error!(target: "panic", "{info}\n{backtrace}");
180        log::logger().flush();
181        previous(info);
182    }));
183    Ok(())
184}

186/// Folder holding the log files, once logging has started.
187pub fn directory() -> Option<&'static Path> {
188    LOGGER.get().map(|logger| logger.directory.as_path())
189}

191/// Returns up to `lines` of the most recent log lines, oldest first, reading into rotated files
192/// when the current one is shorter.
193pub fn tail(lines: usize) -> Result<String, String> {
194    let directory = directory().ok_or_else(|| {
195        "Logging didn't start. Can you check that the data folder is writable?".to_string()
196    })?;
197    tail_in(directory, lines.clamp(1, MAX_TAIL_LINES))
198}
199
200fn tail_in(directory: &Path, lines: usize) -> Result<String, String> {
201    let files = std::iter::once(directory.join(FILE_NAME))
202        .chain((1..=BACKUPS).map(|index| directory.join(format!("{FILE_NAME}.{index}"))));
203    let mut collected: Vec<String> = Vec::new();
204    for path in files {
205        let bytes = match fs::read(&path) {
206            Ok(bytes) => bytes,
207            Err(error) if error.kind() == io::ErrorKind::NotFound => continue,
208            Err(_) => {
209                return Err("The log couldn't be read. Is another program using it?".into());
210            }
211        };
212        let text = String::from_utf8_lossy(&bytes);
213        let mut older: Vec<String> = text
214            .lines()
215            .rev()
216            .take(lines - collected.len())
217            .map(str::to_owned)
218            .collect();
219        collected.append(&mut older);
220        if collected.len() >= lines {
221            break;
222        }
223    }
224    collected.reverse();
225    Ok(collected.join("\n"))
226}

228/// Formats one record as a redacted line: `<UTC time> <LEVEL> <target>: <message>`.
229///
230/// Messages are cut at [`MAX_MESSAGE_BYTES`]. Continuation lines are indented and other control
231/// characters are escaped, so a message can't imitate another record.
232fn format_line(time: SystemTime, level: Level, target: &str, message: &str) -> String {
233    let target = if target == CRATE_ROOT {
234        "toolblox"
235    } else {
236        target.strip_prefix(CRATE_PREFIX).unwrap_or(target)
237    };
238    let mut message = redact(message);
239    if message.len() > MAX_MESSAGE_BYTES {
240        let mut end = MAX_MESSAGE_BYTES;
241        while !message.is_char_boundary(end) {
242            end -= 1;
243        }
244        message.truncate(end);
245        message.push_str(" [truncated]");
246    }
247    let mut line = format!(
248        "{} {:<5} {}: ",
249        timestamp(time),
250        level,
251        sanitize(&redact(target), false)
252    );
253    line.push_str(&sanitize(message.trim_end(), true));
254    line.push('\n');
255    line
256}

258/// Escapes control characters. With `multiline`, line breaks become indented continuation lines.
259fn sanitize(text: &str, multiline: bool) -> String {
260    let mut output = String::with_capacity(text.len());
261    for character in text.chars() {
262        match character {
263            '\n' if multiline => output.push_str("\n    "),
264            '\r' if multiline => {}
265            '\t' => output.push(' '),
266            character if character.is_control() => {
267                let _ = write!(output, "\\u{{{:x}}}", character as u32);
268            }
269            character => output.push(character),
270        }
271    }
272    output
273}

275/// Formats a time as UTC ISO 8601 with milliseconds.
276fn timestamp(time: SystemTime) -> String {
277    let elapsed = time.duration_since(UNIX_EPOCH).unwrap_or_default();
278    let seconds = elapsed.as_secs();
279    let (year, month, day) = civil_from_days((seconds / 86_400) as i64);
280    let remainder = seconds % 86_400;
281    format!(
282        "{year:04}-{month:02}-{day:02}T{:02}:{:02}:{:02}.{:03}Z",
283        remainder / 3_600,
284        remainder % 3_600 / 60,
285        remainder % 60,
286        elapsed.subsec_millis()
287    )
288}

290/// Converts days since 1970-01-01 to a proleptic Gregorian date (Howard Hinnant's algorithm).
291fn civil_from_days(days: i64) -> (i64, u32, u32) {
292    let shifted = days + 719_468;
293    let era = shifted.div_euclid(146_097);
294    let day_of_era = shifted.rem_euclid(146_097);
295    let year_of_era =
296        (day_of_era - day_of_era / 1_460 + day_of_era / 36_524 - day_of_era / 146_096) / 365;
297    let day_of_year = day_of_era - (365 * year_of_era + year_of_era / 4 - year_of_era / 100);
298    let month_index = (5 * day_of_year + 2) / 153;
299    let day = (day_of_year - (153 * month_index + 2) / 5 + 1) as u32;
300    let month = if month_index < 10 {
301        month_index + 3
302    } else {
303        month_index - 9
304    } as u32;
305    (year_of_era + era * 400 + i64::from(month <= 2), month, day)
306}

308/// Replaces Roblox session cookies, launch URIs, and the values of [`SECRET_KEYS`] with
309/// `[redacted]`.
310pub fn redact(text: &str) -> String {
311    let lower = text.to_ascii_lowercase();
312    let bytes = text.as_bytes();
313    let mut output = String::with_capacity(text.len());
314    let mut copied = 0;
315    let mut index = 0;
316    while index < bytes.len() {
317        let Some((value_start, value_end)) = secret_at(&lower, bytes, index) else {
318            index += 1;
319            continue;
320        };
321        output.push_str(&text[copied..value_start]);
322        output.push_str(REDACTED);
323        copied = value_end;
324        index = value_end.max(index + 1);
325    }
326    output.push_str(&text[copied..]);
327    output
328}

330/// Finds a secret starting at `index` and returns the byte range of its value.
331fn secret_at(lower: &str, bytes: &[u8], index: usize) -> Option<(usize, usize)> {
332    if !lower.is_char_boundary(index) {
333        return None;
334    }
335    let rest = &lower[index..];
336    if rest.starts_with(COOKIE_MARKER) {
337        return Some((index, value_end(bytes, index, false)));
338    }
339    let boundary = index == 0 || !is_word_byte(bytes[index - 1]);
340    if !boundary {
341        return None;
342    }
343    if let Some(scheme) = LAUNCH_SCHEMES
344        .iter()
345        .find(|scheme| rest.starts_with(**scheme))
346    {
347        let start = index + scheme.len();
348        let end = value_end(bytes, start, false);
349        return (end > start).then_some((start, end));
350    }
351    let key = SECRET_KEYS
352        .iter()
353        .filter(|key| rest.starts_with(**key))
354        .filter(|key| {
355            bytes
356                .get(index + key.len())
357                .is_none_or(|next| !is_word_byte(*next))
358        })
359        .max_by_key(|key| key.len())?;
360    let mut cursor = index + key.len();
361    if bytes.get(cursor) == Some(&b'"') {
362        cursor += 1;
363    }
364    cursor = skip_spaces(bytes, cursor);
365    if !matches!(bytes.get(cursor), Some(b'=' | b':')) {
366        return None;
367    }
368    cursor = skip_spaces(bytes, cursor + 1);
369    let quoted = bytes.get(cursor) == Some(&b'"');
370    if quoted {
371        cursor += 1;
372    }
373    let end = if quoted {
374        bytes[cursor..]
375            .iter()
376            .position(|byte| *byte == b'"' || *byte == b'\n')
377            .map_or(bytes.len(), |offset| cursor + offset)
378    } else {
379        value_end(bytes, cursor, WHOLE_VALUE_KEYS.contains(key))
380    };
381    (end > cursor && &lower[cursor..end] != REDACTED).then_some((cursor, end))
382}
383
384/// End of an unquoted value: the next whitespace, quote, or separator. With `whole`, only line
385/// breaks and quotes end it.
386fn value_end(bytes: &[u8], start: usize, whole: bool) -> usize {
387    bytes[start..]
388        .iter()
389        .position(|byte| {
390            if whole {
391                matches!(byte, b'\n' | b'\r' | b'"' | b'\'')
392            } else {
393                byte.is_ascii_whitespace()
394                    || matches!(byte, b';' | b',' | b'"' | b'\'' | b'&' | b'<' | b'>' | b'`')
395            }
396        })
397        .map_or(bytes.len(), |offset| start + offset)
398}
399
400fn skip_spaces(bytes: &[u8], mut cursor: usize) -> usize {
401    while bytes.get(cursor) == Some(&b' ') {
402        cursor += 1;
403    }
404    cursor
405}
406
407fn is_word_byte(byte: u8) -> bool {
408    byte.is_ascii_alphanumeric() || matches!(byte, b'_' | b'-' | b'.')
409}
