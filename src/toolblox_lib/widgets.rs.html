41    /// Sandboxed dashboard tile page, which must be `dashboard.html`.
42    #[serde(default)]
43    pub dashboard_entry: Option<String>,
44    #[serde(default)]
45    pub description: String,
46    #[serde(default)]
47    pub icon: Option<String>,
48    #[serde(default)]
49    pub permissions: Vec<WidgetPermission>,

53    /// Executables under `optional-bin/` the widget may start. Requires `process.spawn`.
54    #[serde(default)]

154impl WidgetManifest {
155    /// Checks the schema version, ID, name, SemVer version, fixed entry file names, permissions,
156    /// network origins, and processes.
157    pub fn validate(&self) -> Result<(), String> {
158        if self.schema_version != 1 {
159            return Err("This widget uses an unsupported manifest version.".into());
160        }
161        validate_widget_id(&self.id)?;
162        if self.name.trim().is_empty() || self.name.chars().count() > 80 {
163            return Err("The widget name must contain between 1 and 80 characters.".into());
164        }
165        Version::parse(&self.version).map_err(|_| "The widget version isn't valid SemVer.")?;
166        if self.entry.is_none() && self.module.is_none() {
167            return Err("The widget must declare an entry page or a module.".into());
168        }
169        if self
170            .entry
171            .as_deref()
172            .is_some_and(|entry| entry != "index.html")
173        {
174            return Err("The widget entry must be index.html.".into());
175        }
176        if self
177            .module
178            .as_deref()
179            .is_some_and(|module| module != "main.js")
180        {
181            return Err("The widget module must be main.js.".into());
182        }
183        if self
184            .settings_entry
185            .as_deref()
186            .is_some_and(|entry| entry != "settings.html")
187        {
188            return Err("The widget settings entry must be settings.html.".into());
189        }
190        if self
191            .dashboard_entry
192            .as_deref()
193            .is_some_and(|entry| entry != "dashboard.html")
194        {
195            return Err("The widget dashboard entry must be dashboard.html.".into());
196        }
197        let permissions: HashSet<_> = self.permissions.iter().collect();
198        if permissions.len() != self.permissions.len() {
199            return Err("The widget manifest contains duplicate permissions.".into());
200        }
201        for origin in &self.network_origins {
202            validate_https_origin(origin)?;
203        }
204        if !self.network_origins.is_empty()
205            && !permissions.contains(&WidgetPermission::NetworkFetch)
206        {
207            return Err(
208                "The widget declares network origins without network.fetch permission.".into(),
209            );
210        }
211        let mut process_ids = HashSet::new();
212        for process in &self.processes {
213            validate_process_id(&process.id)?;
214            validate_relative_file(&process.executable, "process executable")?;
215            if !process.executable.starts_with("optional-bin/") {
216                return Err("Widget processes must live under optional-bin/.".into());
217            }
218            if !process_ids.insert(&process.id) {
219                return Err("The widget manifest contains duplicate process IDs.".into());
220            }
221        }
222        if !self.processes.is_empty() && !permissions.contains(&WidgetPermission::ProcessSpawn) {
223            return Err("The widget declares processes without process.spawn permission.".into());
224        }
225        Ok(())
226    }
227}

229impl Catalogue {
230    /// Verifies the catalogue signature, then parses it and validates each entry's ID, version,
231    /// checksum, permissions, and HTTPS archive URL.
232    pub fn parse_verified(
233        bytes: &[u8],
234        signature_base64: &str,
235        public_key_base64: &str,
236    ) -> Result<Self, String> {
237        verify_signature(bytes, signature_base64, public_key_base64)?;
238        let catalogue: Self = serde_json::from_slice(bytes)
239            .map_err(|_| "The widget catalogue isn't valid JSON.".to_string())?;
240        if catalogue.schema_version != 1 {
241            return Err("The widget catalogue version isn't supported.".into());
242        }
243        let mut ids = HashSet::new();
244        for entry in &catalogue.entries {
245            validate_widget_id(&entry.id)?;
246            Version::parse(&entry.version)
247                .map_err(|_| format!("Widget {} has an invalid version.", entry.id))?;
248            validate_sha256(&entry.sha256)?;
249            let permissions: HashSet<_> = entry.permissions.iter().collect();
250            if permissions.len() != entry.permissions.len() {
251                return Err(format!(
252                    "Widget {} has duplicate catalogue permissions.",
253                    entry.id
254                ));
255            }
256            let url = Url::parse(&entry.archive_url)
257                .map_err(|_| format!("Widget {} has an invalid archive URL.", entry.id))?;
258            if url.scheme() != "https" || url.host_str().is_none() {
259                return Err(format!(
260                    "Widget {} must use an HTTPS archive URL.",
261                    entry.id
262                ));
263            }
264            if !ids.insert(&entry.id) {
265                return Err(format!(
266                    "The catalogue contains widget {} more than once.",
267                    entry.id
268                ));
269            }
270        }
271        Ok(catalogue)
272    }
273}

275/// Installs a widget package, replacing any installed version only after every check passes.
276///
277/// Checks the size and SHA-256, extracts into a staging folder with path, count, size, and
278/// symlink limits, requires the manifest's ID, version, and permissions to match the catalogue,
279/// and requires every declared file to exist. The previous version is restored if the swap fails.
280pub fn install_archive(
281    archive: &[u8],
282    expected_id: &str,
283    expected_version: &str,
284    expected_sha256: &str,
285    expected_permissions: &[WidgetPermission],
286    widgets_root: &Path,
287) -> Result<InstalledWidget, String> {
288    validate_widget_id(expected_id)?;
289    validate_sha256(expected_sha256)?;
290    if archive.len() > MAX_ARCHIVE_BYTES {
291        return Err("The widget package is larger than 20 MB.".into());
292    }
293    let actual_sha256 = hex::encode(Sha256::digest(archive));
294    if !actual_sha256.eq_ignore_ascii_case(expected_sha256) {
295        return Err("The widget package checksum doesn't match the catalogue.".into());
296    }
297
298    fs::create_dir_all(widgets_root)
299        .map_err(|_| "Couldn't create the widget directory. Is it writable?".to_string())?;
300    let nonce = Uuid::new_v4();
301    let staging = widgets_root.join(format!(".{expected_id}-install-{nonce}"));
302    let backup = widgets_root.join(format!(".{expected_id}-backup-{nonce}"));
303    let destination = widgets_root.join(expected_id);
304    fs::create_dir(&staging)
305        .map_err(|_| "Couldn't create the widget staging directory. Is it writable?".to_string())?;
306
307    let result = extract_archive(archive, &staging).and_then(|_| {
308        let manifest = read_manifest(&staging)?;
309        if manifest.id != expected_id {
310            return Err("The widget package ID doesn't match the catalogue.".into());
311        }
312        if manifest.version != expected_version {
313            return Err("The widget package version doesn't match the catalogue.".into());
314        }
315        if manifest.permissions.iter().collect::<HashSet<_>>()
316            != expected_permissions.iter().collect::<HashSet<_>>()
317        {
318            return Err("The widget permissions don't match the catalogue.".into());
319        }
320        for entry in [
321            manifest.entry.as_deref(),
322            manifest.module.as_deref(),
323            manifest.settings_entry.as_deref(),
324            manifest.dashboard_entry.as_deref(),
325        ]
326        .into_iter()
327        .flatten()
328        {
329            if !staging.join(entry).is_file() {
330                return Err(format!("The widget package doesn't contain {entry}."));
331            }
332        }
333        for process in &manifest.processes {
334            if !staging.join(&process.executable).is_file() {
335                return Err(format!("Widget process {} is missing.", process.id));
336            }
337        }
338        write_installed_metadata(&staging, &manifest, &actual_sha256)?;
339        replace_atomically(&staging, &destination, &backup)?;
340        Ok(InstalledWidget {
341            id: manifest.id.clone(),
342            version: manifest.version.clone(),
343            sha256: actual_sha256,
344            manifest,
345        })
346    });
347
348    if staging.exists() {
349        let _ = fs::remove_dir_all(&staging);
350    }
351    if result.is_ok() && backup.exists() {
352        let _ = fs::remove_dir_all(&backup);
353    }
354    result
355}

357/// Reads every installed widget, sorted by ID. Hidden folders, such as staging folders, are
358/// skipped, and each unreadable widget yields an error.
359pub fn discover_installed(widgets_root: &Path) -> Vec<Result<InstalledWidget, String>> {
360    let Ok(entries) = fs::read_dir(widgets_root) else {
361        return Vec::new();
362    };
363    let mut found = Vec::new();
364    for entry in entries.flatten() {
365        let name = entry.file_name().to_string_lossy().into_owned();
366        if name.starts_with('.') || !entry.path().is_dir() {
367            continue;
368        }
369        found.push(
370            read_installed(&entry.path())
371                .map_err(|error| format!("Widget {name} couldn't load: {error}")),
372        );
373    }
374    found.sort_by(|left, right| {
375        let left_id = left.as_ref().map(|widget| widget.id.as_str()).unwrap_or("");
376        let right_id = right
377            .as_ref()
378            .map(|widget| widget.id.as_str())
379            .unwrap_or("");
380        left_id.cmp(right_id)
381    });
382    found
383}

385/// Removes an installed widget's folder. A missing widget succeeds.
386pub fn uninstall_widget(widget_id: &str, widgets_root: &Path) -> Result<(), String> {
387    validate_widget_id(widget_id)?;
388    let target = widgets_root.join(widget_id);
389    if !target.exists() {
390        return Ok(());
391    }
392    fs::remove_dir_all(target)
393        .map_err(|_| "Couldn't remove the widget. Is a widget process still running?".to_string())
394}

396/// Extracts a ZIP into `staging`, rejecting unsafe or duplicate paths, symlinks, too many
397/// entries, and oversized content.
398fn extract_archive(archive: &[u8], staging: &Path) -> Result<(), String> {
399    let mut zip = ZipArchive::new(Cursor::new(archive))
400        .map_err(|_| "The widget package isn't a valid ZIP archive.".to_string())?;
401    if zip.len() > MAX_FILES {
402        return Err("The widget package contains too many files.".into());
403    }
404    let mut names = HashSet::new();
405    let mut expanded = 0_u64;
406    for index in 0..zip.len() {
407        let mut item = zip
408            .by_index(index)
409            .map_err(|_| "Couldn't read the widget package.".to_string())?;
410        let enclosed = item
411            .enclosed_name()
412            .ok_or_else(|| "The widget package contains an unsafe path.".to_string())?
413            .to_path_buf();
414        validate_archive_path(&enclosed)?;
415        let normalized = enclosed.to_string_lossy().replace('\\', "/");
416        if !names.insert(normalized) {
417            return Err("The widget package contains duplicate paths.".into());
418        }
419        if item
420            .unix_mode()
421            .is_some_and(|mode| mode & 0o170000 == 0o120000)
422        {
423            return Err("The widget package contains a symbolic link.".into());
424        }
425        expanded = expanded
426            .checked_add(item.size())
427            .ok_or_else(|| "The widget package is too large when extracted.".to_string())?;
428        if expanded > MAX_EXPANDED_BYTES {
429            return Err("The widget package is too large when extracted.".into());
430        }
431        let output = staging.join(&enclosed);
432        if item.is_dir() {
433            fs::create_dir_all(&output)
434                .map_err(|_| "Couldn't create a widget package directory.".to_string())?;
435            continue;
436        }
437        if let Some(parent) = output.parent() {
438            fs::create_dir_all(parent)
439                .map_err(|_| "Couldn't create a widget package directory.".to_string())?;
440        }
441        let mut file = File::create(&output)
442            .map_err(|_| "Couldn't create a widget package file.".to_string())?;
443        std::io::copy(&mut item, &mut file)
444            .map_err(|_| "Couldn't extract the widget package.".to_string())?;
445        file.sync_all()
446            .map_err(|_| "Couldn't save the widget package.".to_string())?;
447        #[cfg(unix)]
448        if let Some(mode) = item.unix_mode() {
449            use std::os::unix::fs::PermissionsExt;
450
451            fs::set_permissions(&output, fs::Permissions::from_mode(mode & 0o777))
452                .map_err(|_| "Couldn't set widget package permissions.".to_string())?;
453        }
454    }
455    Ok(())
456}

458/// Reads and validates `widget.json`, which may be at most 128 KB.
459fn read_manifest(root: &Path) -> Result<WidgetManifest, String> {
460    let bytes = fs::read(root.join("widget.json"))
461        .map_err(|_| "The widget package doesn't contain widget.json.".to_string())?;
462    if bytes.len() > 128 * 1024 {
463        return Err("The widget manifest is too large.".into());
464    }
465    let manifest: WidgetManifest = serde_json::from_slice(&bytes)
466        .map_err(|_| "The widget manifest isn't valid JSON.".to_string())?;
467    manifest.validate()?;
468    Ok(manifest)
469}

471/// Reads an installed widget and checks its install record, including that `widget.json`
472/// hasn't changed since install. Debug builds accept widgets without a record.
473fn read_installed(root: &Path) -> Result<InstalledWidget, String> {
474    let manifest = read_manifest(root)?;
475    let metadata_bytes = match fs::read(root.join(".installed.json")) {
476        Ok(bytes) => bytes,
477        Err(_) if cfg!(debug_assertions) => {
478            return Ok(InstalledWidget {
479                id: manifest.id.clone(),
480                version: manifest.version.clone(),
481                sha256: "0".repeat(64),
482                manifest,
483            });
484        }
485        Err(_) => {
486            return Err(format!(
487                "Widget {} is missing install metadata.",
488                manifest.id
489            ));
490        }
491    };
492    let metadata: InstalledMetadata = serde_json::from_slice(&metadata_bytes)
493        .map_err(|_| format!("Widget {} has invalid install metadata.", manifest.id))?;
494    if metadata.schema_version != 1
495        || metadata.id != manifest.id
496        || metadata.version != manifest.version
497    {
498        return Err(format!(
499            "Widget {} has inconsistent install metadata.",
500            manifest.id
501        ));
502    }
503    validate_sha256(&metadata.sha256)?;
504    validate_sha256(&metadata.manifest_sha256)?;
505    let manifest_bytes = fs::read(root.join("widget.json"))
506        .map_err(|_| format!("Widget {} is missing widget.json.", manifest.id))?;
507    if hex::encode(Sha256::digest(&manifest_bytes)) != metadata.manifest_sha256 {
508        return Err(format!(
509            "Widget {} has a modified manifest. Reinstall it from the Catalogue.",
510            manifest.id
511        ));
512    }
513    Ok(InstalledWidget {
514        id: manifest.id.clone(),
515        version: manifest.version.clone(),
516        sha256: metadata.sha256,
517        manifest,
518    })
519}

521/// Writes `.installed.json` for a staged widget.
522fn write_installed_metadata(
523    root: &Path,
524    manifest: &WidgetManifest,
525    sha256: &str,
526) -> Result<(), String> {
527    let metadata = InstalledMetadata {
528        schema_version: 1,
529        id: manifest.id.clone(),
530        version: manifest.version.clone(),
531        sha256: sha256.to_owned(),
532        manifest_sha256: hex::encode(Sha256::digest(
533            fs::read(root.join("widget.json"))
534                .map_err(|_| "Couldn't read widget.json after installation.".to_string())?,
535        )),
536    };
537    let bytes = serde_json::to_vec_pretty(&metadata)
538        .map_err(|_| "Couldn't encode widget install metadata.".to_string())?;
539    let path = root.join(".installed.json");
540    let mut file =
541        File::create(path).map_err(|_| "Couldn't create widget install metadata.".to_string())?;
542    file.write_all(&bytes)
543        .and_then(|_| file.sync_all())
544        .map_err(|_| "Couldn't save widget install metadata.".to_string())
545}

547/// Moves `staging` to `destination`, keeping the old folder in `backup` and restoring it if
548/// the move fails.
549fn replace_atomically(staging: &Path, destination: &Path, backup: &Path) -> Result<(), String> {
550    if destination.exists() {
551        fs::rename(destination, backup)
552            .map_err(|_| "Couldn't prepare the installed widget for replacement.".to_string())?;
553    }
554    if let Err(error) = fs::rename(staging, destination) {
555        if backup.exists() {
556            let _ = fs::rename(backup, destination);
557        }
558        return Err(format!("Couldn't activate the widget package: {error}"));
559    }
560    Ok(())
561}

563/// Verifies a base64 Ed25519 signature with a base64 raw 32-byte public key.
564fn verify_signature(
565    bytes: &[u8],
566    signature_base64: &str,
567    public_key_base64: &str,
568) -> Result<(), String> {
569    let decoder = base64::engine::general_purpose::STANDARD;
570    let key_bytes = decoder
571        .decode(public_key_base64)
572        .map_err(|_| "The catalogue public key isn't valid.".to_string())?;
573    let signature_bytes = decoder
574        .decode(signature_base64)
575        .map_err(|_| "The catalogue signature isn't valid.".to_string())?;
576    let key_array: [u8; 32] = key_bytes
577        .try_into()
578        .map_err(|_| "The catalogue public key has the wrong length.".to_string())?;
579    let signature = Signature::from_slice(&signature_bytes)
580        .map_err(|_| "The catalogue signature has the wrong length.".to_string())?;
581    let key = VerifyingKey::from_bytes(&key_array)
582        .map_err(|_| "The catalogue public key isn't valid.".to_string())?;
583    key.verify(bytes, &signature)
584        .map_err(|_| "The widget catalogue signature couldn't be verified.".to_string())
585}

587/// Accepts IDs of 1 to 64 lowercase letters, digits, and underscores.
588fn validate_widget_id(value: &str) -> Result<(), String> {
589    if value.is_empty()
590        || value.len() > 64
591        || !value
592            .bytes()
593            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'_')
594    {
595        return Err(
596            "Widget IDs may contain only lowercase letters, numbers, and underscores.".into(),
597        );
598    }
599    Ok(())
600}

602/// Accepts IDs of 1 to 64 ASCII letters, digits, `_`, and `-`.
603fn validate_process_id(value: &str) -> Result<(), String> {
604    if value.is_empty()
605        || value.len() > 64
606        || !value
607            .bytes()
608            .all(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'_' | b'-'))
609    {
610        return Err("Widget process IDs contain unsupported characters.".into());
611    }
612    Ok(())
613}

615/// Requires a non-empty relative path that doesn't leave the package.
616fn validate_relative_file(value: &str, label: &str) -> Result<(), String> {
617    let path = Path::new(value);
618    if value.is_empty()
619        || path.is_absolute()
620        || path.components().any(|part| {
621            matches!(
622                part,
623                Component::ParentDir | Component::RootDir | Component::Prefix(_)
624            )
625        })
626    {
627        return Err(format!("The widget {label} path isn't safe."));
628    }
629    Ok(())
630}

632/// Rejects archive paths with parent, root, or drive components.
633fn validate_archive_path(path: &Path) -> Result<(), String> {
634    if path.components().any(|part| {
635        matches!(
636            part,
637            Component::ParentDir | Component::RootDir | Component::Prefix(_)
638        )
639    }) {
640        return Err("The widget package contains an unsafe path.".into());
641    }
642    Ok(())
643}
644
645/// Accepts bare HTTPS origins, such as `https://example.com`, without a path, query, or
646/// credentials.
647fn validate_https_origin(value: &str) -> Result<(), String> {
648    let url = Url::parse(value).map_err(|_| "A widget network origin isn't valid.".to_string())?;
649    if url.scheme() != "https"
650        || url.host_str().is_none()
651        || url.path() != "/"
652        || url.query().is_some()
653        || url.fragment().is_some()
654        || !url.username().is_empty()
655        || url.password().is_some()
656    {
657        return Err("Widget network origins must be bare HTTPS origins.".into());
658    }
659    Ok(())
660}
661
662/// Requires 64 hex characters.
663fn validate_sha256(value: &str) -> Result<(), String> {
664    if value.len() != 64 || !value.bytes().all(|byte| byte.is_ascii_hexdigit()) {
665        return Err("A widget checksum isn't a valid SHA-256 value.".into());
666    }
667    Ok(())
668}

711        let cursor = Cursor::new(Vec::new());
712        let mut writer = zip::ZipWriter::new(cursor);

714            writer
715                .start_file(*name, SimpleFileOptions::default())
716                .unwrap();

718        }

720    }
