154impl WidgetManifest {
155    /// Checks the schema version, ID, name, SemVer version, fixed entry file names, permissions,
156    /// network origins, and processes.
157    pub fn validate(&self) -> Result<(), String> {
158        if self.schema_version != 1 {
159            return Err("This widget uses an unsupported manifest version.".into());
160        }
161        validate_widget_id(&self.id)?;
162        if self.name.trim().is_empty() || self.name.chars().count() > 80 {
163            return Err("The widget name must contain between 1 and 80 characters.".into());
164        }
165        Version::parse(&self.version).map_err(|_| "The widget version isn't valid SemVer.")?;
166        if self.entry.is_none() && self.module.is_none() {
167            return Err("The widget must declare an entry page or a module.".into());
168        }
169        if self
170            .entry
171            .as_deref()
172            .is_some_and(|entry| entry != "index.html")
173        {
174            return Err("The widget entry must be index.html.".into());
175        }
176        if self
177            .module
178            .as_deref()
179            .is_some_and(|module| module != "main.js")
180        {
181            return Err("The widget module must be main.js.".into());
182        }
183        if self
184            .settings_entry
185            .as_deref()
186            .is_some_and(|entry| entry != "settings.html")
187        {
188            return Err("The widget settings entry must be settings.html.".into());
189        }
190        if self
191            .dashboard_entry
192            .as_deref()
193            .is_some_and(|entry| entry != "dashboard.html")
194        {
195            return Err("The widget dashboard entry must be dashboard.html.".into());
196        }
197        let permissions: HashSet<_> = self.permissions.iter().collect();
198        if permissions.len() != self.permissions.len() {
199            return Err("The widget manifest contains duplicate permissions.".into());
200        }
201        for origin in &self.network_origins {
202            validate_https_origin(origin)?;
203        }
204        if !self.network_origins.is_empty()
205            && !permissions.contains(&WidgetPermission::NetworkFetch)
206        {
207            return Err(
208                "The widget declares network origins without network.fetch permission.".into(),
209            );
210        }
211        let mut process_ids = HashSet::new();
212        for process in &self.processes {
213            validate_process_id(&process.id)?;
214            validate_relative_file(&process.executable, "process executable")?;
215            if !process.executable.starts_with("optional-bin/") {
216                return Err("Widget processes must live under optional-bin/.".into());
217            }
218            if !process_ids.insert(&process.id) {
219                return Err("The widget manifest contains duplicate process IDs.".into());
220            }
221        }
222        if !self.processes.is_empty() && !permissions.contains(&WidgetPermission::ProcessSpawn) {
223            return Err("The widget declares processes without process.spawn permission.".into());
224        }
225        Ok(())
226    }
227}

229impl Catalogue {
230    /// Verifies the catalogue signature, then parses it and validates each entry's ID, version,
231    /// checksum, permissions, and HTTPS archive URL.
232    pub fn parse_verified(
233        bytes: &[u8],
234        signature_base64: &str,
235        public_key_base64: &str,
236    ) -> Result<Self, String> {
237        verify_signature(bytes, signature_base64, public_key_base64)?;
238        let catalogue: Self = serde_json::from_slice(bytes)
239            .map_err(|_| "The widget catalogue isn't valid JSON.".to_string())?;
240        if catalogue.schema_version != 1 {
241            return Err("The widget catalogue version isn't supported.".into());
242        }
243        let mut ids = HashSet::new();
244        for entry in &catalogue.entries {
245            validate_widget_id(&entry.id)?;
246            Version::parse(&entry.version)
247                .map_err(|_| format!("Widget {} has an invalid version.", entry.id))?;
248            validate_sha256(&entry.sha256)?;
249            let permissions: HashSet<_> = entry.permissions.iter().collect();
250            if permissions.len() != entry.permissions.len() {
251                return Err(format!(
252                    "Widget {} has duplicate catalogue permissions.",
253                    entry.id
254                ));
255            }
256            let url = Url::parse(&entry.archive_url)
257                .map_err(|_| format!("Widget {} has an invalid archive URL.", entry.id))?;
258            if url.scheme() != "https" || url.host_str().is_none() {
259                return Err(format!(
260                    "Widget {} must use an HTTPS archive URL.",
261                    entry.id
262                ));
263            }
264            if !ids.insert(&entry.id) {
265                return Err(format!(
266                    "The catalogue contains widget {} more than once.",
267                    entry.id
268                ));
269            }
270        }
271        Ok(catalogue)
272    }
273}

275/// Installs a widget package, replacing any installed version only after every check passes.
276///
277/// Checks the size and SHA-256, extracts into a staging folder with path, count, size, and
278/// symlink limits, requires the manifest's ID, version, and permissions to match the catalogue,
279/// and requires every declared file to exist. The previous version is restored if the swap fails.
280pub fn install_archive(
281    archive: &[u8],
282    expected_id: &str,
283    expected_version: &str,
284    expected_sha256: &str,
285    expected_permissions: &[WidgetPermission],
286    widgets_root: &Path,
287) -> Result<InstalledWidget, String> {
288    validate_widget_id(expected_id)?;
289    validate_sha256(expected_sha256)?;
290    if archive.len() > MAX_ARCHIVE_BYTES {
291        return Err("The widget package is larger than 20 MB.".into());
292    }
293    let actual_sha256 = hex::encode(Sha256::digest(archive));
294    if !actual_sha256.eq_ignore_ascii_case(expected_sha256) {
295        return Err("The widget package checksum doesn't match the catalogue.".into());
296    }
297
298    fs::create_dir_all(widgets_root)
299        .map_err(|_| "Couldn't create the widget directory. Is it writable?".to_string())?;
300    let nonce = Uuid::new_v4();
301    let staging = widgets_root.join(format!(".{expected_id}-install-{nonce}"));
302    let backup = widgets_root.join(format!(".{expected_id}-backup-{nonce}"));
303    let destination = widgets_root.join(expected_id);
304    fs::create_dir(&staging)
305        .map_err(|_| "Couldn't create the widget staging directory. Is it writable?".to_string())?;
306
307    let result = extract_archive(archive, &staging).and_then(|_| {
308        let manifest = read_manifest(&staging)?;
309        if manifest.id != expected_id {
310            return Err("The widget package ID doesn't match the catalogue.".into());
311        }
312        if manifest.version != expected_version {
313            return Err("The widget package version doesn't match the catalogue.".into());
314        }
315        if manifest.permissions.iter().collect::<HashSet<_>>()
316            != expected_permissions.iter().collect::<HashSet<_>>()
317        {
318            return Err("The widget permissions don't match the catalogue.".into());
319        }
320        for entry in [
321            manifest.entry.as_deref(),
322            manifest.module.as_deref(),
323            manifest.settings_entry.as_deref(),
324            manifest.dashboard_entry.as_deref(),
325        ]
326        .into_iter()
327        .flatten()
328        {
329            if !staging.join(entry).is_file() {
330                return Err(format!("The widget package doesn't contain {entry}."));
331            }
332        }
333        for process in &manifest.processes {
334            if !staging.join(&process.executable).is_file() {
335                return Err(format!("Widget process {} is missing.", process.id));
336            }
337        }
338        write_installed_metadata(&staging, &manifest, &actual_sha256)?;
339        replace_atomically(&staging, &destination, &backup)?;
340        Ok(InstalledWidget {
341            id: manifest.id.clone(),
342            version: manifest.version.clone(),
343            sha256: actual_sha256,
344            manifest,
345        })
346    });
347
348    if staging.exists() {
349        let _ = fs::remove_dir_all(&staging);
350    }
351    if result.is_ok() && backup.exists() {
352        let _ = fs::remove_dir_all(&backup);
353    }
354    result
355}

357/// Reads every installed widget, sorted by ID. Hidden folders, such as staging folders, are
358/// skipped, and each unreadable widget yields an error.
359pub fn discover_installed(widgets_root: &Path) -> Vec<Result<InstalledWidget, String>> {
360    let Ok(entries) = fs::read_dir(widgets_root) else {
361        return Vec::new();
362    };
363    let mut found = Vec::new();
364    for entry in entries.flatten() {
365        let name = entry.file_name().to_string_lossy().into_owned();
366        if name.starts_with('.') || !entry.path().is_dir() {
367            continue;
368        }
369        found.push(
370            read_installed(&entry.path())
371                .map_err(|error| format!("Widget {name} couldn't load: {error}")),
372        );
373    }
374    found.sort_by(|left, right| {
375        let left_id = left.as_ref().map(|widget| widget.id.as_str()).unwrap_or("");
376        let right_id = right
377            .as_ref()
378            .map(|widget| widget.id.as_str())
379            .unwrap_or("");
380        left_id.cmp(right_id)
381    });
382    found
383}

402/// Adds the example widget to the widgets folder once. A hidden marker file records that it was
403/// added, so an uninstalled example stays uninstalled and an edited copy is never replaced.
404/// Development builds only.
405#[cfg(debug_assertions)]
406pub fn add_example_widget(widgets_root: &Path) -> Result<(), String> {
407    let marker = widgets_root.join(".example-added");
408    let target = widgets_root.join("hello_world");
409    if marker.exists() {
410        return Ok(());
411    }
412    if !target.exists() {
413        fs::create_dir_all(&target)
414            .map_err(|_| "Couldn't create the example widget folder.".to_string())?;
415        for (name, contents) in EXAMPLE_WIDGET {
416            fs::write(target.join(name), contents)
417                .map_err(|_| format!("Couldn't write the example widget's {name}."))?;
418        }
419    }
420    fs::write(marker, "").map_err(|_| "Couldn't record the example widget.".to_string())
421}

431/// Builds an unsigned catalogue from the widget folders in `roots`, packaging each one in memory
432/// so installs run the same checks as catalogue packages. Missing folders list no widgets.
433/// Development builds only.
434#[cfg(debug_assertions)]
435pub fn local_catalogue(roots: &[&Path]) -> Result<Catalogue, String> {
436    let mut folders: Vec<_> = roots
437        .iter()
438        .filter_map(|root| fs::read_dir(root).ok())
439        .flat_map(|entries| entries.flatten())
440        .filter(|entry| {
441            entry.path().is_dir() && !entry.file_name().to_string_lossy().starts_with('.')
442        })
443        .map(|entry| entry.path())
444        .collect();
445    folders.sort();
446    let mut entries: Vec<CatalogueEntry> = Vec::new();
447    for folder in folders {
448        let name = folder
449            .file_name()
450            .unwrap_or_default()
451            .to_string_lossy()
452            .into_owned();
453        let manifest = read_manifest(&folder).map_err(|error| format!("Widget {name}: {error}"))?;
454        if manifest.id != name {
455            return Err(format!(
456                "Widget {name}: the manifest ID must match its folder name."
457            ));
458        }
459        if entries.iter().any(|entry| entry.id == name) {
460            return Err(format!(
461                "The catalogue contains widget {name} more than once."
462            ));
463        }
464        let archive = package_folder(&folder)?;
465        entries.push(CatalogueEntry {
466            id: manifest.id,
467            name: manifest.name,
468            version: manifest.version,
469            description: manifest.description,
470            archive_url: Url::from_directory_path(&folder)
471                .map_err(|_| format!("Widget {name}: the folder path isn't absolute."))?
472                .to_string(),
473            sha256: hex::encode(Sha256::digest(&archive)),
474            permissions: manifest.permissions,
475            icon_url: None,
476        });
477    }
478    Ok(Catalogue {
479        schema_version: 1,
480        generated_at: 0,
481        entries,
482    })
483}

485/// Packages a widget folder as a ZIP, skipping hidden files and rejecting symbolic links.
486/// Development builds only.
487#[cfg(debug_assertions)]
488pub fn package_folder(folder: &Path) -> Result<Vec<u8>, String> {
489    fn collect(folder: &Path, files: &mut Vec<std::path::PathBuf>) -> Result<(), String> {
490        let entries =
491            fs::read_dir(folder).map_err(|_| "Couldn't read a widget folder.".to_string())?;
492        for entry in entries.flatten() {
493            if entry.file_name().to_string_lossy().starts_with('.') {
494                continue;
495            }
496            let kind = entry
497                .file_type()
498                .map_err(|_| "Couldn't read a widget file.".to_string())?;
499            if kind.is_symlink() {
500                return Err("The widget folder contains a symbolic link.".into());
501            }
502            if kind.is_dir() {
503                collect(&entry.path(), files)?;
504            } else {
505                files.push(entry.path());
506            }
507        }
508        Ok(())
509    }
510    let mut files = Vec::new();
511    collect(folder, &mut files)?;
512    files.sort();
513    let mut zip = zip::ZipWriter::new(Cursor::new(Vec::new()));
514    for file in files {
515        let name = file
516            .strip_prefix(folder)
517            .map_err(|_| "Couldn't read a widget file.".to_string())?
518            .to_string_lossy()
519            .replace('\\', "/");
520        let options = zip::write::SimpleFileOptions::default()
521            .compression_method(zip::CompressionMethod::Deflated);
522        #[cfg(unix)]
523        let options = {
524            use std::os::unix::fs::PermissionsExt;
525            let mode = fs::metadata(&file)
526                .map_err(|_| "Couldn't read a widget file.".to_string())?
527                .permissions()
528                .mode();
529            options.unix_permissions(mode & 0o777)
530        };
531        zip.start_file(name, options)
532            .and_then(|()| {
533                fs::read(&file)
534                    .and_then(|bytes| zip.write_all(&bytes))
535                    .map_err(Into::into)
536            })
537            .map_err(|_| "Couldn't package a widget file.".to_string())?;
538    }
539    zip.finish()
540        .map(Cursor::into_inner)
541        .map_err(|_| "Couldn't package the widget.".to_string())
542}

555/// Extracts a ZIP into `staging`, rejecting unsafe or duplicate paths, symlinks, too many
556/// entries, and oversized content.
557fn extract_archive(archive: &[u8], staging: &Path) -> Result<(), String> {
558    let mut zip = ZipArchive::new(Cursor::new(archive))
559        .map_err(|_| "The widget package isn't a valid ZIP archive.".to_string())?;
560    if zip.len() > MAX_FILES {
561        return Err("The widget package contains too many files.".into());
562    }
563    let mut names = HashSet::new();
564    let mut expanded = 0_u64;
565    for index in 0..zip.len() {
566        let mut item = zip
567            .by_index(index)
568            .map_err(|_| "Couldn't read the widget package.".to_string())?;
569        let enclosed = item
570            .enclosed_name()
571            .ok_or_else(|| "The widget package contains an unsafe path.".to_string())?
572            .to_path_buf();
573        validate_archive_path(&enclosed)?;
574        let normalized = enclosed.to_string_lossy().replace('\\', "/");
575        if !names.insert(normalized) {
576            return Err("The widget package contains duplicate paths.".into());
577        }
578        if item
579            .unix_mode()
580            .is_some_and(|mode| mode & 0o170000 == 0o120000)
581        {
582            return Err("The widget package contains a symbolic link.".into());
583        }
584        expanded = expanded
585            .checked_add(item.size())
586            .ok_or_else(|| "The widget package is too large when extracted.".to_string())?;
587        if expanded > MAX_EXPANDED_BYTES {
588            return Err("The widget package is too large when extracted.".into());
589        }
590        let output = staging.join(&enclosed);
591        if item.is_dir() {
592            fs::create_dir_all(&output)
593                .map_err(|_| "Couldn't create a widget package directory.".to_string())?;
594            continue;
595        }
596        if let Some(parent) = output.parent() {
597            fs::create_dir_all(parent)
598                .map_err(|_| "Couldn't create a widget package directory.".to_string())?;
599        }
600        let mut file = File::create(&output)
601            .map_err(|_| "Couldn't create a widget package file.".to_string())?;
602        std::io::copy(&mut item, &mut file)
603            .map_err(|_| "Couldn't extract the widget package.".to_string())?;
604        file.sync_all()
605            .map_err(|_| "Couldn't save the widget package.".to_string())?;
606        #[cfg(unix)]
607        if let Some(mode) = item.unix_mode() {
608            use std::os::unix::fs::PermissionsExt;
609
610            fs::set_permissions(&output, fs::Permissions::from_mode(mode & 0o777))
611                .map_err(|_| "Couldn't set widget package permissions.".to_string())?;
612        }
613    }
614    Ok(())
615}

617/// Reads and validates `widget.json`, which may be at most 128 KB.
618fn read_manifest(root: &Path) -> Result<WidgetManifest, String> {
619    let bytes = fs::read(root.join("widget.json"))
620        .map_err(|_| "The widget package doesn't contain widget.json.".to_string())?;
621    if bytes.len() > 128 * 1024 {
622        return Err("The widget manifest is too large.".into());
623    }
624    let manifest: WidgetManifest = serde_json::from_slice(&bytes)
625        .map_err(|_| "The widget manifest isn't valid JSON.".to_string())?;
626    manifest.validate()?;
627    Ok(manifest)
628}

630/// Reads an installed widget and checks its install record, including that `widget.json`
631/// hasn't changed since install. Debug builds accept widgets without a record.
632fn read_installed(root: &Path) -> Result<InstalledWidget, String> {
633    let manifest = read_manifest(root)?;
634    let metadata_bytes = match fs::read(root.join(".installed.json")) {
635        Ok(bytes) => bytes,
636        Err(_) if cfg!(debug_assertions) => {
637            return Ok(InstalledWidget {
638                id: manifest.id.clone(),
639                version: manifest.version.clone(),
640                sha256: "0".repeat(64),
641                manifest,
642            });
643        }
644        Err(_) => {
645            return Err(format!(
646                "Widget {} is missing install metadata.",
647                manifest.id
648            ));
649        }
650    };
651    let metadata: InstalledMetadata = serde_json::from_slice(&metadata_bytes)
652        .map_err(|_| format!("Widget {} has invalid install metadata.", manifest.id))?;
653    if metadata.schema_version != 1
654        || metadata.id != manifest.id
655        || metadata.version != manifest.version
656    {
657        return Err(format!(
658            "Widget {} has inconsistent install metadata.",
659            manifest.id
660        ));
661    }
662    validate_sha256(&metadata.sha256)?;
663    validate_sha256(&metadata.manifest_sha256)?;
664    let manifest_bytes = fs::read(root.join("widget.json"))
665        .map_err(|_| format!("Widget {} is missing widget.json.", manifest.id))?;
666    if hex::encode(Sha256::digest(&manifest_bytes)) != metadata.manifest_sha256 {
667        return Err(format!(
668            "Widget {} has a modified manifest. Reinstall it from the Catalogue.",
669            manifest.id
670        ));
671    }
672    Ok(InstalledWidget {
673        id: manifest.id.clone(),
674        version: manifest.version.clone(),
675        sha256: metadata.sha256,
676        manifest,
677    })
678}

680/// Writes `.installed.json` for a staged widget.
681fn write_installed_metadata(
682    root: &Path,
683    manifest: &WidgetManifest,
684    sha256: &str,
685) -> Result<(), String> {
686    let metadata = InstalledMetadata {
687        schema_version: 1,
688        id: manifest.id.clone(),
689        version: manifest.version.clone(),
690        sha256: sha256.to_owned(),
691        manifest_sha256: hex::encode(Sha256::digest(
692            fs::read(root.join("widget.json"))
693                .map_err(|_| "Couldn't read widget.json after installation.".to_string())?,
694        )),
695    };
696    let bytes = serde_json::to_vec_pretty(&metadata)
697        .map_err(|_| "Couldn't encode widget install metadata.".to_string())?;
698    let path = root.join(".installed.json");
699    let mut file =
700        File::create(path).map_err(|_| "Couldn't create widget install metadata.".to_string())?;
701    file.write_all(&bytes)
702        .and_then(|_| file.sync_all())
703        .map_err(|_| "Couldn't save widget install metadata.".to_string())
704}

706/// Moves `staging` to `destination`, keeping the old folder in `backup` and restoring it if
707/// the move fails.
708fn replace_atomically(staging: &Path, destination: &Path, backup: &Path) -> Result<(), String> {
709    if destination.exists() {
710        fs::rename(destination, backup)
711            .map_err(|_| "Couldn't prepare the installed widget for replacement.".to_string())?;
712    }
713    if let Err(error) = fs::rename(staging, destination) {
714        if backup.exists() {
715            let _ = fs::rename(backup, destination);
716        }
717        return Err(format!("Couldn't activate the widget package: {error}"));
718    }
719    Ok(())
720}

722/// Verifies a base64 Ed25519 signature with a base64 raw 32-byte public key.
723fn verify_signature(
724    bytes: &[u8],
725    signature_base64: &str,
726    public_key_base64: &str,
727) -> Result<(), String> {
728    let decoder = base64::engine::general_purpose::STANDARD;
729    let key_bytes = decoder
730        .decode(public_key_base64)
731        .map_err(|_| "The catalogue public key isn't valid.".to_string())?;
732    let signature_bytes = decoder
733        .decode(signature_base64)
734        .map_err(|_| "The catalogue signature isn't valid.".to_string())?;
735    let key_array: [u8; 32] = key_bytes
736        .try_into()
737        .map_err(|_| "The catalogue public key has the wrong length.".to_string())?;
738    let signature = Signature::from_slice(&signature_bytes)
739        .map_err(|_| "The catalogue signature has the wrong length.".to_string())?;
740    let key = VerifyingKey::from_bytes(&key_array)
741        .map_err(|_| "The catalogue public key isn't valid.".to_string())?;
742    key.verify(bytes, &signature)
743        .map_err(|_| "The widget catalogue signature couldn't be verified.".to_string())
744}

746/// Accepts IDs of 1 to 64 lowercase letters, digits, and underscores.
747pub(crate) fn validate_widget_id(value: &str) -> Result<(), String> {
748    if value.is_empty()
749        || value.len() > 64
750        || !value
751            .bytes()
752            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'_')
753    {
754        return Err(
755            "Widget IDs may contain only lowercase letters, numbers, and underscores.".into(),
756        );
757    }
758    Ok(())
759}

761/// Accepts IDs of 1 to 64 ASCII letters, digits, `_`, and `-`.
762fn validate_process_id(value: &str) -> Result<(), String> {
763    if value.is_empty()
764        || value.len() > 64
765        || !value
766            .bytes()
767            .all(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'_' | b'-'))
768    {
769        return Err("Widget process IDs contain unsupported characters.".into());
770    }
771    Ok(())
772}

774/// Requires a non-empty relative path that doesn't leave the package.
775fn validate_relative_file(value: &str, label: &str) -> Result<(), String> {
776    let path = Path::new(value);
777    if value.is_empty()
778        || path.is_absolute()
779        || path.components().any(|part| {
780            matches!(
781                part,
782                Component::ParentDir | Component::RootDir | Component::Prefix(_)
783            )
784        })
785    {
786        return Err(format!("The widget {label} path isn't safe."));
787    }
788    Ok(())
789}

791/// Rejects archive paths with parent, root, or drive components.
792fn validate_archive_path(path: &Path) -> Result<(), String> {
793    if path.components().any(|part| {
794        matches!(
795            part,
796            Component::ParentDir | Component::RootDir | Component::Prefix(_)
797        )
798    }) {
799        return Err("The widget package contains an unsafe path.".into());
800    }
801    Ok(())
802}

804/// Accepts bare HTTPS origins, such as `https://example.com`, without a path, query, or
805/// credentials.
806fn validate_https_origin(value: &str) -> Result<(), String> {
807    let url = Url::parse(value).map_err(|_| "A widget network origin isn't valid.".to_string())?;
808    if url.scheme() != "https"
809        || url.host_str().is_none()
810        || url.path() != "/"
811        || url.query().is_some()
812        || url.fragment().is_some()
813        || !url.username().is_empty()
814        || url.password().is_some()
815    {
816        return Err("Widget network origins must be bare HTTPS origins.".into());
817    }
818    Ok(())
819}

821/// Requires 64 hex characters.
822fn validate_sha256(value: &str) -> Result<(), String> {
823    if value.len() != 64 || !value.bytes().all(|byte| byte.is_ascii_hexdigit()) {
824        return Err("A widget checksum isn't a valid SHA-256 value.".into());
825    }
826    Ok(())
827}
