3use std::fmt;

6/// Keychain and Credential Manager service name for Roblox sessions.
7const SERVICE: &str = "Toolblox";
8
9/// Start of every Roblox session cookie value, in lowercase.
10pub(crate) const SESSION_COOKIE_MARKER: &str = "_|warning:";
11
12/// Whether text contains a Roblox session cookie value, ignoring case.
13pub(crate) fn contains_session_cookie(text: &str) -> bool {
14    text.to_ascii_lowercase().contains(SESSION_COOKIE_MARKER)
15}
16
17/// A Roblox session cookie. Its `Debug` output is redacted and its memory is zeroed on drop.
18pub struct Secret(String);
19
20impl Secret {
21    /// Wraps a session value read from storage or a sign-in.
22    pub(crate) fn new(value: String) -> Self {
23        Self(value)
24    }
25
26    /// Returns the raw session value. Callers must not log or return it to JavaScript.
27    pub(crate) fn expose(&self) -> &str {
28        &self.0
29    }
30}
31
32impl fmt::Debug for Secret {
33    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
34        formatter.write_str("Secret([redacted])")
35    }
36}
37
38impl Drop for Secret {
39    fn drop(&mut self) {
40        self.0.zeroize();
41    }
42}

44/// Why a session couldn't be read or written.
45#[derive(Debug)]
46pub enum CredentialError {
47    /// No session is stored for the account, or an empty session was given.
48    Missing,
49    /// This platform has no supported secure storage.
50    #[cfg(not(any(windows, target_os = "macos")))]
51    Unsupported,
52    /// The operating system's secure storage failed or is locked.
53    Platform,
54}
55
56impl fmt::Display for CredentialError {
57    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
58        match self {
59            Self::Missing => write!(
60                formatter,
61                "The saved session wasn't found. Can you sign in again?"
62            ),
63            #[cfg(not(any(windows, target_os = "macos")))]
64            Self::Unsupported => write!(
65                formatter,
66                "Secure session storage isn't supported here. Can you use Windows or macOS?"
67            ),
68            Self::Platform => write!(
69                formatter,
70                "Toolblox couldn't access secure session storage. Is it unlocked?"
71            ),
72        }
73    }
74}
75
76impl std::error::Error for CredentialError {}

78/// Stores one Roblox session per account ID.
79pub trait CredentialStore: Send + Sync {
80    /// Reads the account's session.
81    fn get(&self, account_id: u64) -> Result<Secret, CredentialError>;
82    /// Stores or replaces the account's session. Rejects an empty session.
83    fn set(&self, account_id: u64, secret: &str) -> Result<(), CredentialError>;
84    /// Deletes the account's session. Deleting a missing session succeeds.
85    fn delete(&self, account_id: u64) -> Result<(), CredentialError>;
86}
87
88/// Stores sessions in Windows Credential Manager or the macOS login Keychain.
89#[derive(Debug, Default)]
90pub struct OsCredentialStore;
91
92impl OsCredentialStore {
93    /// Opens the keyring entry for an account.
94    #[cfg(any(windows, target_os = "macos"))]
95    fn entry(account_id: u64) -> Result<keyring::Entry, CredentialError> {
96        keyring::Entry::new(SERVICE, &format!("account-{account_id}"))
97            .map_err(|_| CredentialError::Platform)
98    }
99}
100
101#[cfg(any(windows, target_os = "macos"))]
102impl CredentialStore for OsCredentialStore {
103    fn get(&self, account_id: u64) -> Result<Secret, CredentialError> {
104        Self::entry(account_id)?
105            .get_password()
106            .map(Secret)
107            .map_err(|error| match error {
108                keyring::Error::NoEntry => CredentialError::Missing,
109                _ => CredentialError::Platform,
110            })
111    }
112
113    fn set(&self, account_id: u64, secret: &str) -> Result<(), CredentialError> {
114        if secret.is_empty() {
115            return Err(CredentialError::Missing);
116        }
117        Self::entry(account_id)?
118            .set_password(secret)
119            .map_err(|_| CredentialError::Platform)
120    }
121
122    fn delete(&self, account_id: u64) -> Result<(), CredentialError> {
123        match Self::entry(account_id)?.delete_credential() {
124            Ok(()) | Err(keyring::Error::NoEntry) => Ok(()),
125            Err(_) => Err(CredentialError::Platform),
126        }
127    }
128}
129
130#[cfg(not(any(windows, target_os = "macos")))]
131impl CredentialStore for OsCredentialStore {
132    fn get(&self, _account_id: u64) -> Result<Secret, CredentialError> {
133        Err(CredentialError::Unsupported)
134    }
135
136    fn set(&self, _account_id: u64, _secret: &str) -> Result<(), CredentialError> {
137        Err(CredentialError::Unsupported)
138    }
139
140    fn delete(&self, _account_id: u64) -> Result<(), CredentialError> {
141        Err(CredentialError::Unsupported)
142    }
143}

171    /// Every place a session secret is read from the platform or exposed as text, with the files
172    /// allowed to do it and how often. A new entry needs security review: it must not return the
173    /// secret to JavaScript, a widget, a log, or a non-Roblox host.
174    const SECRET_ACCESS: &[(&str, &[(&str, usize)])] = &[
175        (".expose()", &[("app_commands.rs", 1), ("roblox.rs", 1)]),
176        (".expose_secret()", &[("app_commands.rs", 2)]),
177        ("cookie_header(", &[("roblox.rs", 4)]),
178        ("keyring::", &[("credentials.rs", 4)]),
179        ("get_password", &[("credentials.rs", 1)]),
180        (".cookies()", &[("login.rs", 1)]),
181        ("find-generic-password", &[]),
182        ("CryptUnprotectData", &[]),
183    ];
184
185    /// Returns a source file's non-test code with comment lines removed.
186    pub(crate) fn production_source(source: &str) -> String {
187        let source = source.replace("\r\n", "\n");
188        let end = ["#[cfg(test)]\nmod ", "#[cfg(test)]\npub(crate) mod "]
189            .iter()
190            .filter_map(|marker| source.find(marker))
191            .min()
192            .unwrap_or(source.len());
193        source[..end]
194            .lines()
195            .filter(|line| !line.trim_start().starts_with("//"))
196            .collect::<Vec<_>>()
197            .join("\n")
198    }
199
200    #[test]
201    fn production_source_excludes_tests_with_either_line_ending() {
202        for visibility in ["", "pub(crate) "] {
203            let source = format!(
204                "// Comment\nfn production() {{}}\n#[cfg(test)]\n{visibility}mod tests {{}}\n"
205            );
206            let expected = "fn production() {}";
207            assert_eq!(production_source(&source), expected);
208            assert_eq!(production_source(&source.replace('\n', "\r\n")), expected);
209        }
210    }

212    #[test]
213    fn secret_access_points_match_the_reviewed_list() {
214        let directory = Path::new(env!("CARGO_MANIFEST_DIR")).join("src");
215        let mut files: Vec<_> = std::fs::read_dir(&directory)
216            .unwrap()
217            .map(|entry| entry.unwrap().path())
218            .filter(|path| path.extension().is_some_and(|extension| extension == "rs"))
219            .collect();
220        files.sort();
221        assert!(!files.is_empty());
222        let mut problems = Vec::new();
223        for (pattern, allowed) in SECRET_ACCESS {
224            for path in &files {
225                let name = path.file_name().unwrap().to_str().unwrap();
226                let source = std::fs::read_to_string(path).unwrap();
227                let found = production_source(&source).matches(pattern).count();
228                let expected = allowed
229                    .iter()
230                    .find(|(file, _)| *file == name)
231                    .map_or(0, |(_, count)| *count);
232                if found != expected {
233                    problems.push(format!(
234                        "{name} uses `{pattern}` {found} times; the reviewed count is {expected}"
235                    ));
236                }
237            }
238        }
239        assert!(
240            problems.is_empty(),
241            "Session secret access changed. Review that the secret can't reach JavaScript, a \
242             widget, a log, or a non-Roblox host, then update SECRET_ACCESS.\n{}",
243            problems.join("\n")
244        );
245    }
