15/// Scheduling priority for a Roblox client.
16#[derive(Debug, Clone, Copy, Default, Serialize, Deserialize, PartialEq, Eq)]
17#[serde(rename_all = "snake_case")]
18pub enum ClientPriority {
19    #[default]
20    Normal,
21    BelowNormal,
22    Low,
23}

25/// Window size used when tiling Roblox clients.
26#[derive(Debug, Clone, Copy, Default, Serialize, Deserialize, PartialEq, Eq)]
27#[serde(rename_all = "snake_case")]
28pub enum TileSize {
29    /// Divide the screen evenly between the windows.

35    Small,

38/// Largest CPU core limit accepted per client.
39pub const MAX_CPU_CORES: u32 = 64;

41/// Seconds to watch for a launched client's process.
42const WATCH_SECONDS: u32 = 30;

44/// How a launched client is tuned. The default leaves the process unchanged.
45#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
46pub struct Tuning {
47    pub priority: ClientPriority,
48    /// Logical cores the client may use. `0` allows every core.
49    pub cpu_cores: u32,
50    /// Size of the client's window once it opens. Windows only.
51    pub window_size: Option<WindowSize>,
52}
53
54impl Tuning {
55    /// Whether the process needs to be changed at all.
56    pub fn changes_process(&self) -> bool {
57        *self != Self::default()
58    }
59}

61/// Tracks which Roblox processes Toolblox already tuned, so batch launches tune each new client
62/// once.
63#[derive(Debug, Default)]
64pub struct ClientTuner {
65    /// Process IDs claimed by a launch, with the account they belong to.
66    claimed: Mutex<HashMap<u32, u64>>,
67    /// Rotates which cores each limited client gets.
68    next_slot: AtomicUsize,
69}
70
71impl ClientTuner {
72    /// Claims a Roblox process that started after `before` and isn't claimed yet.
73    fn claim(&self, before: &HashSet<u32>, running: &HashSet<u32>, account_id: u64) -> Option<u32> {
74        let mut claimed = self.claimed.lock().ok()?;
75        claimed.retain(|pid, _| running.contains(pid));
76        let pid = running
77            .iter()
78            .copied()
79            .filter(|pid| !before.contains(pid) && !claimed.contains_key(pid))
80            .min()?;
81        claimed.insert(pid, account_id);
82        Some(pid)
83    }
84
85    /// Watches for the client a launch starts and tunes it. If Roblox replaces its first process,
86    /// as it does while updating, the replacement is tuned too.
87    ///
88    /// ponytail: launches in a batch are matched to new processes in the order they appear. Two
89    /// clients that start out of order swap tuning. Matching by command line needs process APIs
90    /// Toolblox doesn't use yet.
91    pub async fn watch(&self, before: HashSet<u32>, tuning: Tuning, account_id: u64) {
92        let slot = self.next_slot.fetch_add(1, Ordering::Relaxed);
93        let mask = affinity_mask(slot, tuning.cpu_cores, available_cores());
94        let mut current = None;
95        let mut sized = tuning.window_size.is_none();
96        for _ in 0..WATCH_SECONDS {
97            tokio::time::sleep(std::time::Duration::from_secs(1)).await;
98            let Ok(Ok(running)) = tauri::async_runtime::spawn_blocking(roblox_pids).await else {
99                continue;
100            };
101            if !current.is_some_and(|pid| running.contains(&pid)) {
102                current = self.claim(&before, &running, account_id);
103                sized = tuning.window_size.is_none();
104                if let Some(pid) = current
105                    && (tuning.priority != ClientPriority::Normal || mask.is_some())
106                {
107                    match apply(pid, tuning.priority, mask) {
108                        Ok(()) => log::info!("Tuned Roblox client {pid} for account {account_id}"),
109                        Err(error) => log::warn!("Roblox client {pid} couldn't be tuned: {error}"),
110                    }
111                }
112            }
113            if let (Some(pid), Some(size), false) = (current, tuning.window_size, sized) {
114                sized = resize_client(pid, size);
115            }
116        }
117        if current.is_none() {
118            log::info!("No new Roblox client appeared for account {account_id}");
119        }
120    }
121}

123/// Seconds to wait for a launched client before the next launch.
124const START_TIMEOUT_SECONDS: u32 = 20;
125/// Seconds a new client gets to read Roblox's files before the next launch rewrites them.
126const READ_FILES_SECONDS: u64 = 3;
127
128/// Waits until a Roblox client that wasn't in `before` is running, then a little longer, so the
129/// next launch can rewrite Roblox's files. Gives up after 20 seconds.
130///
131/// ponytail: the grace period is fixed. A client that reads its files later than that starts with
132/// the next account's settings. Watching Roblox's log for the point it reads them would be exact.
133pub async fn wait_for_client(before: &HashSet<u32>) {
134    for _ in 0..START_TIMEOUT_SECONDS * 2 {
135        tokio::time::sleep(std::time::Duration::from_millis(500)).await;
136        let started = tauri::async_runtime::spawn_blocking(roblox_pids)
137            .await
138            .ok()
139            .and_then(Result::ok)
140            .is_some_and(|running| running.iter().any(|pid| !before.contains(pid)));
141        if started {
142            tokio::time::sleep(std::time::Duration::from_secs(READ_FILES_SECONDS)).await;
143            return;
144        }
145    }
146    log::info!("No new Roblox client appeared before the next launch");
147}

149/// Number of logical cores, capped at the 64 one affinity mask can hold.
150fn available_cores() -> u32 {
151    std::thread::available_parallelism()
152        .map_or(1, |count| count.get() as u32)
153        .min(usize::BITS)
154}
155
156/// The cores client number `slot` may use: `cores` consecutive cores, starting where the previous
157/// client's left off. Returns `None` when every core is allowed.
158pub(crate) fn affinity_mask(slot: usize, cores: u32, available: u32) -> Option<usize> {
159    if cores == 0 || cores >= available {
160        return None;
161    }
162    let start = (slot * cores as usize) % available as usize;
163    Some((0..cores as usize).fold(0, |mask, offset| {
164        mask | 1 << ((start + offset) % available as usize)
165    }))
166}

168/// Process IDs of the running Roblox clients.
169pub fn roblox_pids() -> Result<HashSet<u32>, String> {
170    #[cfg(windows)]
171    {
172        use std::os::windows::process::CommandExt;
173
174        const CREATE_NO_WINDOW: u32 = 0x08000000;
175        let output = std::process::Command::new("tasklist.exe")
176            .args([
177                "/FI",
178                "IMAGENAME eq RobloxPlayerBeta.exe",
179                "/FO",
180                "CSV",
181                "/NH",
182            ])
183            .creation_flags(CREATE_NO_WINDOW)
184            .output()
185            .map_err(|_| {
186                "Toolblox couldn't inspect running Roblox processes. Can you try again?".to_string()
187            })?;
188        Ok(parse_tasklist(&String::from_utf8_lossy(&output.stdout)))
189    }
190    #[cfg(target_os = "macos")]
191    {
192        let output = std::process::Command::new("/usr/bin/pgrep")
193            .args(["-x", "RobloxPlayer"])
194            .output()
195            .map_err(|_| {
196                "Toolblox couldn't inspect running Roblox processes. Can you try again?".to_string()
197            })?;
198        Ok(String::from_utf8_lossy(&output.stdout)
199            .lines()
200            .filter_map(|line| line.trim().parse().ok())
201            .collect())
202    }
203    #[cfg(not(any(windows, target_os = "macos")))]
204    {
205        Ok(HashSet::new())
206    }
207}

209/// Reads process IDs from `tasklist /FO CSV /NH` output.
210#[cfg_attr(not(windows), allow(dead_code))]
211fn parse_tasklist(text: &str) -> HashSet<u32> {
212    text.lines()
213        .filter_map(|line| line.split("\",\"").nth(1))
214        .filter_map(|value| value.trim_matches('"').parse().ok())
215        .collect()
216}

218/// Sets a client's priority and, on Windows, the cores it may use.
219fn apply(pid: u32, priority: ClientPriority, mask: Option<usize>) -> Result<(), String> {
220    #[cfg(windows)]
221    {
222        use windows_sys::Win32::Foundation::CloseHandle;
223        use windows_sys::Win32::System::Threading::{
224            BELOW_NORMAL_PRIORITY_CLASS, IDLE_PRIORITY_CLASS, OpenProcess, PROCESS_SET_INFORMATION,
225            SetPriorityClass, SetProcessAffinityMask,
226        };
227
228        // SAFETY: `OpenProcess` takes no pointers; a null result is handled.
229        let process = unsafe { OpenProcess(PROCESS_SET_INFORMATION, 0, pid) };
230        if process.is_null() {
231            return Err("The Roblox client couldn't be opened. Has it closed?".into());
232        }
233        let class = match priority {
234            ClientPriority::Normal => None,
235            ClientPriority::BelowNormal => Some(BELOW_NORMAL_PRIORITY_CLASS),
236            ClientPriority::Low => Some(IDLE_PRIORITY_CLASS),
237        };
238        // SAFETY: `process` is an open handle with set-information access, closed once below.
239        let applied = unsafe {
240            class.is_none_or(|class| SetPriorityClass(process, class) != 0)
241                && mask.is_none_or(|mask| SetProcessAffinityMask(process, mask) != 0)
242        };
243        // SAFETY: `process` came from `OpenProcess` and isn't used again.
244        unsafe { CloseHandle(process) };
245        if applied {
246            Ok(())
247        } else {
248            Err("The Roblox client's priority couldn't be changed. Has it closed?".into())
249        }
250    }
251    #[cfg(target_os = "macos")]
252    {
253        unsafe extern "C" {
254            fn setpriority(
255                which: std::ffi::c_int,
256                who: u32,
257                priority: std::ffi::c_int,
258            ) -> std::ffi::c_int;
259        }
260        const PRIO_PROCESS: std::ffi::c_int = 0;
261        let _ = mask;
262        let nice = match priority {
263            ClientPriority::Normal => return Ok(()),
264            ClientPriority::BelowNormal => 5,
265            ClientPriority::Low => 15,
266        };
267        // SAFETY: `setpriority` takes no pointers.
268        if unsafe { setpriority(PRIO_PROCESS, pid, nice) } == 0 {
269            Ok(())
270        } else {
271            Err("The Roblox client's priority couldn't be changed. Has it closed?".into())
272        }
273    }
274    #[cfg(not(any(windows, target_os = "macos")))]
275    {
276        let _ = (pid, priority, mask);
277        Err("Changing client priority isn't supported on this platform.".into())
278    }
279}

281/// A screen rectangle in physical pixels.
282#[derive(Debug, Clone, Copy, PartialEq, Eq)]
283pub(crate) struct Rect {
284    pub x: i32,
285    pub y: i32,
286    pub width: i32,
287    pub height: i32,
288}
289
290/// Offset between layers when fixed-size windows don't all fit on the screen.
291const LAYER_OFFSET: i32 = 32;
292
293/// Positions for `count` windows tiled in `area`, left to right and top to bottom.
294pub(crate) fn tile_rects(count: usize, area: Rect, size: TileSize) -> Vec<Rect> {
295    if count == 0 || area.width <= 0 || area.height <= 0 {
296        return Vec::new();
297    }
298    let (width, height) = match size {
299        TileSize::Fit => {
300            let columns = (count as f64).sqrt().ceil() as usize;
301            let rows = count.div_ceil(columns);
302            (area.width / columns as i32, area.height / rows as i32)
303        }
304        TileSize::Medium => (960.min(area.width), 540.min(area.height)),
305        TileSize::Small => (640.min(area.width), 360.min(area.height)),
306    };
307    let columns = (area.width / width).max(1) as usize;
308    let rows = (area.height / height).max(1) as usize;
309    (0..count)
310        .map(|index| {
311            let layer = (index / (columns * rows)) as i32;
312            let cell = index % (columns * rows);
313            Rect {
314                x: area.x + (cell % columns) as i32 * width + layer * LAYER_OFFSET,
315                y: area.y + (cell / columns) as i32 * height + layer * LAYER_OFFSET,
316                width,
317                height,
318            }
319        })
320        .collect()
321}

323/// The visible, unowned top-level windows of the given processes, sorted by process ID.
324#[cfg(windows)]
325fn client_windows(pids: HashSet<u32>) -> Vec<(u32, windows_sys::Win32::Foundation::HWND)> {
326    use windows_sys::Win32::Foundation::{HWND, LPARAM};
327    use windows_sys::Win32::UI::WindowsAndMessaging::{
328        EnumWindows, GW_OWNER, GetWindow, GetWindowThreadProcessId, IsWindowVisible,
329    };
330
331    struct Search {
332        pids: HashSet<u32>,
333        windows: Vec<(u32, HWND)>,
334    }
335
336    unsafe extern "system" fn visit(window: HWND, search: LPARAM) -> i32 {
337        // SAFETY: `search` is the `Search` passed to `EnumWindows`, alive for the whole call.
338        let search = unsafe { &mut *(search as *mut Search) };
339        let mut pid = 0;
340        // SAFETY: `window` comes from `EnumWindows` and `pid` is a valid out pointer.
341        unsafe {
342            GetWindowThreadProcessId(window, &mut pid);
343            if search.pids.contains(&pid)
344                && IsWindowVisible(window) != 0
345                && GetWindow(window, GW_OWNER).is_null()
346            {
347                search.windows.push((pid, window));
348            }
349        }
350        1
351    }
352
353    let mut search = Search {
354        pids,
355        windows: Vec::new(),
356    };
357    if !search.pids.is_empty() {
358        // SAFETY: `visit` only casts the `LPARAM` back to `search`, which outlives the call.
359        unsafe { EnumWindows(Some(visit), &mut search as *mut Search as LPARAM) };
360    }
361    search.windows.sort_by_key(|(pid, _)| *pid);
362    search.windows
363}

365/// Restores a window from minimized or maximized, then gives it the rectangle. With
366/// `keep_position`, only the size changes.
367#[cfg(windows)]
368fn place_window(window: windows_sys::Win32::Foundation::HWND, rect: Rect, keep_position: bool) {
369    use windows_sys::Win32::UI::WindowsAndMessaging::{
370        IsIconic, IsZoomed, SW_RESTORE, SWP_NOACTIVATE, SWP_NOMOVE, SWP_NOZORDER, SetWindowPos,
371        ShowWindow,
372    };
373
374    let flags = SWP_NOZORDER | SWP_NOACTIVATE | if keep_position { SWP_NOMOVE } else { 0 };
375    // SAFETY: `window` is a top-level window found by `client_windows`. A window that closed since
376    // then makes these calls fail harmlessly.
377    unsafe {
378        if IsIconic(window) != 0 || IsZoomed(window) != 0 {
379            ShowWindow(window, SW_RESTORE);
380        }
381        SetWindowPos(
382            window,
383            std::ptr::null_mut(),
384            rect.x,
385            rect.y,
386            rect.width,
387            rect.height,
388            flags,
389        );
390    }
391}

393/// Sizes a client's window. Returns whether the window was found. Does nothing outside Windows.
394fn resize_client(pid: u32, size: WindowSize) -> bool {
395    #[cfg(windows)]
396    {
397        let Some((_, window)) = client_windows(HashSet::from([pid])).into_iter().next() else {
398            return false;
399        };
400        let rect = Rect {
401            x: 0,
402            y: 0,
403            width: size.width,
404            height: size.height,
405        };
406        place_window(window, rect, true);
407        true
408    }
409    #[cfg(not(windows))]
410    {
411        let _ = (pid, size);
412        true
413    }
414}

416/// Tiles every visible Roblox client window on the primary screen and returns how many moved.
417pub fn tile_windows(size: TileSize) -> Result<usize, String> {
418    #[cfg(windows)]
419    {
420        use windows_sys::Win32::Foundation::RECT;
421        use windows_sys::Win32::UI::WindowsAndMessaging::{SPI_GETWORKAREA, SystemParametersInfoW};
422
423        let windows = client_windows(roblox_pids()?);
424        if windows.is_empty() {
425            return Ok(0);
426        }
427        let mut work = RECT {
428            left: 0,
429            top: 0,
430            right: 0,
431            bottom: 0,
432        };
433        // SAFETY: `SPI_GETWORKAREA` writes one `RECT` to the pointer.
434        let found = unsafe {
435            SystemParametersInfoW(SPI_GETWORKAREA, 0, (&mut work as *mut RECT).cast(), 0)
436        };
437        if found == 0 {
438            return Err("The screen size couldn't be read. Can you try again?".into());
439        }
440        let area = Rect {
441            x: work.left,
442            y: work.top,
443            width: work.right - work.left,
444            height: work.bottom - work.top,
445        };
446        let rects = tile_rects(windows.len(), area, size);
447        for ((_, window), rect) in windows.iter().zip(&rects) {
448            place_window(*window, *rect, false);
449        }
450        Ok(rects.len())
451    }
452    #[cfg(not(windows))]
453    {
454        let _ = size;
455        Err("Tiling windows isn't supported on macOS yet.".into())
456    }
457}
